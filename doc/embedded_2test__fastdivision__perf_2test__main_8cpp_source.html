86 v_result16 = (uint16_t)(i & 0xFF) / 10;
87 }
88 endTime = micros();
89 stdTime = endTime - startTime;

257
258 // divu10 timing
259 startTime = micros();
260 for (uint32_t i = 0; i < CYCLE_ITERATIONS; ++i)
261 {
262 v_result16 = divu10(12345);
263 }
264 endTime = micros();
265 uint32_t divu10Time = endTime - startTime;
266
267 // divmod10 timing
268 uint32_t div;
269 uint8_t mod;
270 startTime = micros();
271 for (uint32_t i = 0; i < CYCLE_ITERATIONS; ++i)

273 divmod10(12345678UL, div, mod);
274 v_result32 = div;
275 }
276 endTime = micros();
277 uint32_t divmod10Time = endTime - startTime;
278
279 // Calculate approximate cycles (16MHz = 16 cycles per microsecond)
280 float divu10Cycles = (float)divu10Time * 16.0f / (float)CYCLE_ITERATIONS;
281 float divmod10Cycles = (float)divmod10Time * 16.0f / (float)CYCLE_ITERATIONS;
282
283 Serial.print(F("\n divu10: ~"));
284 Serial.print(divu10Cycles, 1);

287 Serial.print(F("\n divmod10: ~"));

289 Serial.print(F(" cycles"));

292 benchReport(F("fastdivision: divmod10 x1000"), divmod10Time, F("us"));
