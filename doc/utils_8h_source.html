269inline bool printJsonStep(Print& out, uint8_t step)
270{
271 if (step == 0)
272 {
273 out.print(F("{\"P\":"));

276 if constexpr (RELAY_DIVERSION)
277 {
278 out.print(F(",\"R\":"));
279 out.print(relayDiverter.get_average());
280 }
281 return true;
282 }
283 --step;
284
285 if (step < NO_OF_PHASES)
286 {
287 out.print(F(",\"P"));
288 out.print(step + 1);
289 out.print(F("\":"));
290 out.print(tx_data.power_L[step]);
291 return true;
292 }
293 step -= NO_OF_PHASES;
294
295 if constexpr (TEMP_SENSOR_PRESENT)
296 {
297 if (step < tempSensors.size())
298 {
299 if (isValidTemperature(tx_data.temperature_x100[step]))
300 {
301 out.print(F(",\"T"));
302 out.print(step + 1);
303 out.print(F("\":"));
304 SerialOutput::printHundredths(out, tx_data.temperature_x100[step]);
305 }
306 return true;
307 }
308 step -= tempSensors.size();
309 }
310
311 if constexpr (DUAL_TARIFF)
312 {
313 if (step == 0)
314 {
315 out.print(datalogOffPeak ? F(",\"TA\":\"low\"") : F(",\"TA\":\"high\""));
316 return true;
317 }
318 }
319
320 out.println('}');
321 return false;

339inline bool printTextStep(Print& out, uint8_t step)
340{
341 if (step == 0)
342 {
343 SerialOutput::printDecimals(out, Energy::toHundredthsOfJoule< SUPPLY_FREQUENCY >(Shared::copyOf_energyInBucket_main), 2);

345 }
346 if (step == 1)
347 {
348 out.print(F(", P:"));

350
351 if constexpr (RELAY_DIVERSION)
352 {
353 out.print(F("/"));
354 out.print(relayDiverter.get_average());
355 }
356 return true;
357 }
358 step -= 2;
359
360 if (step < NO_OF_PHASES)
361 {
362 out.print(F(", P"));
363 out.print(step + 1);
364 out.print(F(":"));
365 out.print(tx_data.power_L[step]);
366 return true;
367 }
368 step -= NO_OF_PHASES;
369
370 if (step < NO_OF_PHASES)
371 {
372 out.print(F(", V"));
373 out.print(step + 1);
374 out.print(F(":"));
375 SerialOutput::printDecimals(out, tx_data.Vrms_L_x100[step], 2);
376 return true;
377 }
378 step -= NO_OF_PHASES;
379
380 if constexpr (TEMP_SENSOR_PRESENT)
381 {
382 if (step < tempSensors.size())
383 {
384 if (isValidTemperature(tx_data.temperature_x100[step]))
385 {
386 out.print(F(", T"));
387 out.print(step + 1);
388 out.print(F(":"));
389 SerialOutput::printDecimals(out, tx_data.temperature_x100[step], 2);
390 }
391 return true;
392 }
393 step -= tempSensors.size();
394 }
395
396 switch (step)
397 {
398 case 0:
399 out.print(F(", (minSampleSets/MC "));
400 out.print(Shared::copyOf_lowestNoOfSampleSetsPerMainsCycle);
401 return true;
402 case 1:
403 out.print(F(", #ofSampleSets "));
404 out.print(Shared::copyOf_sampleSetsDuringThisDatalogPeriod);
405 return true;
406 case 2:
407 if constexpr (!DUAL_TARIFF && PRIORITY_ROTATION != RotationModes::OFF)
408 {
409 out.print(F(", NoED "));
410 out.print(Shared::absenceOfDivertedEnergyCountInSeconds);
411 }
412 return true;
413 default:
414 out.println(F(")"));
415 return false;
416 }

557inline void sendResults(bool bOffPeak)
558{
559 static bool startup{ true };
560
561 if constexpr (RF_CHIP_PRESENT)
562 {
563 initialize_rf(); // returns at once if the module is up, else retries (about 50 ms when it fails)
564 }
565
566 if (startup)
567 {
568 startup = false;
569 return; // reject the first datalogging which is incomplete !
570 }
571
572 if constexpr (RF_LOGGING_PRESENT)
573 {
574 send_rf_data(tx_data); // *SEND RF DATA*
575 }
576
577 // The output is only started here, and written by SerialOutput::poll() from the
578 // main loop. If the previous one is still in progress (it normally takes a small
579 // part of a datalog period), this one is skipped.
580 if (SerialOutput::busy())
581 {
582 return;
583 }
584
585 if constexpr (SERIAL_OUTPUT_TYPE == SerialOutputType::HumanReadable)
586 {
587 SerialOutput::start(printTextStep);
588 }
589 else if constexpr (SERIAL_OUTPUT_TYPE == SerialOutputType::IoT)
590 {
591 datalogOffPeak = bOffPeak;
592 TeleInfoData::snapshot();
593 SerialOutput::start(writeTeleInfoStep);
594 }
595 else if constexpr (SERIAL_OUTPUT_TYPE == SerialOutputType::JSON)
596 {
597 datalogOffPeak = bOffPeak;
598 SerialOutput::start(printJsonStep);
599 }
600}
