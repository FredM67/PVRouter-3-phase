21// ----------------------------------------------------------------------------
22// An output that records the switches
23// ----------------------------------------------------------------------------
24struct Switch
25{
26 uint8_t pin;
27 bool on;
28};
29
30Switch switches[16];
31uint8_t switchCount{ 0 };
32
33struct Recorder
34{
35 static void switched(const uint8_t pin, const bool on)
36 {
37 if (switchCount < sizeof(switches) / sizeof(switches[0]))
38 {
39 switches[switchCount] = { pin, on };
40 }
41 ++switchCount;
42 }
43};
44
45using Relay = RelayLogic::Relay< Recorder >;
46
47// a 1-minute sliding average: 12 datalogs of 5 s, quick to settle
48template< uint8_t N >
49using Engine = RelayLogic::Engine< Relay, N, 1 >;
50
51constexpr uint16_t NO_OVERRIDE{ 0 };
52
53constexpr uint16_t forced(const uint8_t pin)
54{
55 return static_cast< uint16_t >(1U << pin);
56}
57
58void assertSwitch(const uint8_t i, const uint8_t pin, const bool on)
59{
60 TEST_ASSERT_TRUE_MESSAGE(i < switchCount, "switch not recorded");
61 TEST_ASSERT_EQUAL_UINT8(pin, switches[i].pin);
62 TEST_ASSERT_EQUAL(on, switches[i].on);
63}

89template< uint8_t N >
90void seconds(const Engine< N >& e, uint16_t n, const uint16_t overrideBitmask = NO_OVERRIDE, const bool diversionEnabled = true)
91{
92 while (n--)
93 {
94 e.inc_duration();
95 e.proceed_relays(overrideBitmask, diversionEnabled);
96 }
97}

100template< uint8_t N >
101void skipStartUp(const Engine< N >& e)
102{
103 for (uint8_t i = 0; i < 60; ++i)
104 {
105 e.inc_duration();
106 }
107}

110template< uint8_t N >
111void allOn(const Engine< N >& e)
112{
113 setAverage< N >(-10000);
114 skipStartUp(e);
115 seconds(e, 60 * N);
116 for (uint8_t i = 0; i < 60; ++i)
117 {
118 e.inc_duration();
119 }
120 for (uint8_t i = 0; i < N; ++i)
121 {
122 TEST_ASSERT_TRUE(e.get_relay(i).isRelayON());
123 }
124 switchCount = 0;
125}
