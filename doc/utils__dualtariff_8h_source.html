161constexpr uint32_t windowStart(const BoostWindow& window, const uint8_t offPeakHours)
162{
163 const uint32_t minutes{ window.start.fromEnd ? (static_cast< uint32_t >(offPeakHours) * 60U - window.start.minutes) : window.start.minutes };

172constexpr uint32_t windowEnd(const BoostWindow& window, const uint8_t offPeakHours)
173{
174 if (UNTIL_THE_END == window.length.minutes)
