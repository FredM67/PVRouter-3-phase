50struct Cycles
51{
52 uint16_t min{ UINT16_MAX };
53 uint16_t max{ 0 };
54 uint32_t sum{ 0 };
55
56 void add(uint16_t cycles)
57 {
58 if (cycles < min)
59 {
60 min = cycles;
61 }
62 if (cycles > max)
63 {
64 max = cycles;
65 }
66 sum += cycles;
67 }
68 uint16_t average() const
69 {
70 return static_cast< uint16_t >(sum / N_INPUTS);
71 }
72};

74// Timer1 at the CPU clock: one count per cycle, overflows after 4 ms
75static void startTimer()
76{
77 TCCR1A = 0;
78 TCCR1B = 0;
79 TCNT1 = 0;
80 TCCR1B = _BV(CS10);
81}

83// The code between the two reads of TCNT1, with the overhead of the reads subtracted
84#define MEASURE(stats, overhead, code) \
85 do \
86 { \
87 cli(); \
88 startTimer(); \
89 const uint16_t t0{ TCNT1 }; \
90 code; \
91 const uint16_t t1{ TCNT1 }; \
92 sei(); \
93 (stats).add(static_cast< uint16_t >(t1 - t0 - (overhead))); \
94 } while (0)

96static uint16_t overhead()
97{
98 Cycles empty;
99 for (uint8_t i = 0; i < N_INPUTS; ++i)
100 {
101 MEASURE(empty, 0, {});
102 }
103 return empty.min;
104}

138void test_isqrt_on_avr_every_boundary(void)
139{
140 uint32_t errors{ 0 };
141 uint32_t firstError{ 0 };
142 for (uint32_t root = 0; root <= UINT16_MAX; ++root)
143 {
144 const uint32_t square{ root * root };
145 const auto r{ static_cast< uint16_t >(root) };
146 const uint16_t expected[]{ r, r, r, static_cast< uint16_t >(root < UINT16_MAX ? root + 1 : UINT16_MAX) };
147 const uint32_t values[]{ root > 1 ? square - 1 : square, square, square + root, square + root + 1 }; // sqrt(1 - 1) = 0
148 const uint8_t count{ static_cast< uint8_t >(root < UINT16_MAX ? 4 : 3) }; // 65535^2 + 65536 = 2^32 wraps
149 for (uint8_t i = 0; i < count; ++i)
150 {
151 if ((Datalog::isqrt(values[i]) != expected[i]) && !errors++)
152 {
153 firstError = values[i];
154 }
155 }
156 }
157 if (errors)
158 {
159 Serial.print(F("first wrong value: "));
160 Serial.println(firstError);
161 }
162 TEST_ASSERT_EQUAL_UINT32(0, errors);
163}

181void test_vrms_on_avr_matches_the_float_code(void)
182{
183 uint8_t bits;
184 for (uint32_t meanSquare = 1; meanSquare <= 512UL * 512UL; meanSquare = meanSquare * 9 / 8 + 1)
185 {
186 // sqrtScaled: sqrt(x) x 2^bits within 0.5 of the exact value (+ the float's own rounding)
187 const uint16_t root{ Datalog::sqrtScaled(meanSquare, bits) };
188 const float exact{ sqrtf(static_cast< float >(meanSquare)) * static_cast< float >(1UL << bits) };
189 TEST_ASSERT_FLOAT_WITHIN(0.5F + exact * 1.2e-7F, exact, root);
190
191 // vrms: within 1 hundredth of the float code (which truncated)
192 inMeanSquare = meanSquare;
193 const auto reference{ static_cast< uint32_t >(100U * voltageCal[0] * sqrt(inMeanSquare)) };
194 TEST_ASSERT_UINT32_WITHIN(1, reference, Datalog::vrms(inMeanSquare, voltageFixed.value[0], voltageFixed.shift));
195 }
196}

202void test_isqrt_is_faster_than_float_sqrt(void)
203{
204 const uint16_t ovh{ overhead() };
205 Cycles floatSqrt, intSqrt, scaledSqrt;
206
207 for (uint8_t i = 0; i < N_INPUTS; ++i)
208 {
209 inMeanSquare = meanSquares[i];
210 MEASURE(floatSqrt, ovh, out32 = static_cast< uint32_t >(sqrt(inMeanSquare)));
211 MEASURE(intSqrt, ovh, out32 = Datalog::isqrt(inMeanSquare));
212 MEASURE(scaledSqrt, ovh, {
213 uint8_t bits;
214 out32 = Datalog::sqrtScaled(inMeanSquare, bits);
215 out16 = bits;
216 });
217 }
218
219 Serial.println(F("--- square root ---"));
220 report(F("float sqrt() "), floatSqrt);
221 report(F("Datalog::isqrt() "), intSqrt);
222 report(F("Datalog::sqrtScaled "), scaledSqrt);
223 bench(F("datalog: isqrt avg"), F("datalog: isqrt max"), intSqrt);
224 bench(F("datalog: sqrtScaled avg"), F("datalog: sqrtScaled max"), scaledSqrt);
225
226 TEST_ASSERT_LESS_OR_EQUAL_UINT16(floatSqrt.max, scaledSqrt.max);
227 TEST_ASSERT_LESS_OR_EQUAL_UINT32(floatSqrt.sum, scaledSqrt.sum);
228}
