254 void schedule(uint8_t unitIdx, uint8_t bitmask)
255 {
256 auto &unit{ unitStates[unitIdx] };
257
258 unit.tx_data = bitmask;
259
260 if (bitmask != unit.previousBitmask)
261 {
262 // state changed - transmit on the next pass through loop()
263 unit.previousBitmask = bitmask;
264 unit.cyclesSinceLastUpdate = 0;
265 unit.pendingTransmission = true;
266
267 return;
268 }
269
270 // state unchanged - refresh periodically
271 if (++unit.cyclesSinceLastUpdate >= REMOTE_REFRESH_CYCLES)
272 {
273 unit.cyclesSinceLastUpdate = 0;
274 unit.pendingTransmission = true;
275 }
