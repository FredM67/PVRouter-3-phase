61inline OverrideMasks pinForcing(const Pins& pins, IsActive isActive)
62{
63 OverrideMasks masks{ 0, 0 };
64
65 if (0 == pins.size())
66 {
67 return masks;
68 }
69
70 // a do-while, as before the move: a for loop costs 22 bytes more on the AVR
71 uint8_t i{ pins.size() };
72 do
73 {
74 --i;
75 if (isActive(pins.getPin(i)))
76 {
77 masks.local |= pins.getLocalBitmask(i);
78 masks.remote |= pins.getRemoteBitmask(i);
79 }
80 } while (i);
81 return masks;
82}
