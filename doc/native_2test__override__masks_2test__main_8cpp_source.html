34// Override pins: D3 forces D5 and remote load 1, D4 forces D5 and D6, D12 forces remote load 0

39// Boosts, over 8 hours of off-peak:
40// #0 the first 2 h: D5
41// #1 from 1 h, for 2 h (1 h to 3 h): D6 and remote load 2
42// #2 the last 3 h (5 h to the end): D7
43constexpr BoostSchedule boosts{ { { FIRST(HOURS(2)), { 5 } },
44 { FROM_START(HOURS(1)), FOR(HOURS(2)), { 6, remote(2) } },
45 { LAST(HOURS(3)), { 7 } } } };
