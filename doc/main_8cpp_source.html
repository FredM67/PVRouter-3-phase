82OverrideMasks getDualTariffForcingBitmask(const bool bOffPeak, const int16_t currentTemperature_x100)
83{
84 if constexpr (!DUAL_TARIFF)
85 {
86 return { 0, 0 };
87 }
88
89 constexpr int16_t iTemperatureThreshold_x100{ iTemperatureThreshold * 100 };
90
91 // Return early if we're not in off-peak period
92 if (!bOffPeak)
93 {
94 return { 0, 0 };
95 }
96
97 // We're in off-peak period - check forcing time windows
98 const auto ulElapsedTime{ static_cast< uint32_t >(millis() - ul_TimeOffPeak) };
99
100 OverrideMasks forcing{ 0, 0 };
101 uint8_t i{ NO_OF_DUMPLOADS };
102 do
103 {
104 --i;
105 // Skip if not within the 'force period'
106 if ((ulElapsedTime < rg_OffsetForce[i][0]) || (ulElapsedTime >= rg_OffsetForce[i][1]))
107 {
108 continue;
109 }
110
111 // Force load ON if temperature condition is met
112 if (currentTemperature_x100 <= iTemperatureThreshold_x100)
113 {
114 const uint8_t pin{ overridePinOf(physicalLoadPin, i) };
115
116 if (pin < REMOTE_PIN_BASE)
117 {
118 bit_set(forcing.local, pin);
119 }
120 else
121 {
122 bit_set(forcing.remote, pin - REMOTE_PIN_BASE);
123 }
124 }
125 } while (i);
126
127 return forcing;
128}

152OverrideMasks getOverrideBitmask(const bool bOffPeak, const int16_t currentTemperature_x100)
153{
154 OverrideMasks masks{ 0, 0 };
155
156 // Add external override pins
157 if constexpr (OVERRIDE_PIN_PRESENT)
158 {
159 // Check each configured override pin and combine their bitmasks
160 uint8_t i{ overridePins.size() };
161 do
162 {
163 --i;
164 const uint8_t pin = overridePins.getPin(i);
165 const auto pinState = getPinState(pin);
166
167 if (!pinState) // Pin is LOW (active)
168 {
169 masks.local |= overridePins.getLocalBitmask(i);
170 masks.remote |= overridePins.getRemoteBitmask(i);
171 }
172 } while (i);
173 }
174
175 // Add dual tariff forcing - OR operation handles precedence automatically
176 // If a bit is already set by external override, OR won't change it
177 // If a bit is not set, OR will apply dual tariff forcing
178 const auto forcing{ getDualTariffForcingBitmask(bOffPeak, currentTemperature_x100) };
179 masks.local |= forcing.local;
180 masks.remote |= forcing.remote;
181
182 return masks;
183}

299bool proceedDualTariffLogic()
300{
301 static bool pinOffPeakState{ HIGH };
302 const auto pinNewState{ getPinState(dualTariffPin) };
303
304 if (pinOffPeakState && !pinNewState)
305 {
306 // we start off-peak period
307 statusln(F("Change to off-peak period!"));
308
309 ul_TimeOffPeak = millis();
310
311 if constexpr (PRIORITY_ROTATION == RotationModes::AUTO)
312 {
313 proceedRotation();
314 }
315 }
316
317 // end of off-peak period
318 if (!pinOffPeakState && pinNewState)
319 {
320 statusln(F("Change to peak period!"));
321 }
322
323 pinOffPeakState = pinNewState;
324
325 return (LOW == pinOffPeakState);
326}

435void updatePowerAndVoltageData()
436{
437 // the calibration constants in fixed point, converted at compile time: no float code (#120)
438 static constexpr auto powerCal PROGMEM{ Datalog::powerCalibration(f_powerCal) };
439 static constexpr auto voltageCal PROGMEM{ Datalog::voltageCalibration(Datalog::vrmsFactor(DATALOG_PERIOD_IN_SECONDS), f_voltageCal) };
440
441 tx_data.power = 0;
442 uint8_t phase{ NO_OF_PHASES };
443 do
444 {
445 --phase;
446 const uint16_t sampleSets{ Shared::copyOf_sampleSetsDuringThisDatalogPeriod };
447
448 // import = +ve: the sum of the ISR is negative when importing
449 tx_data.power_L[phase] = Datalog::watts(-Shared::copyOf_sumP_atSupplyPoint[phase] / sampleSets,
450 pgm_read_word(&powerCal.value[phase]), powerCal.shift);
451
452 tx_data.power += tx_data.power_L[phase];
453
454 tx_data.Vrms_L_x100[phase] = Datalog::vrms(Shared::copyOf_sum_Vsquared[phase] / sampleSets,
455 pgm_read_word(&voltageCal.value[phase]), voltageCal.shift);
456 } while (phase);
457}

476{

481

485 tmp = DEVICE_DISCONNECTED_RAW;
486 }
487
488 tx_data.temperature_x100[idx] = tmp;
489 } while (idx);

518void handlePerSecondTasks(bool &bOffPeak, int16_t &iTemperature_x100)
519{
520 if constexpr (WATCHDOG_PIN_PRESENT)
521 {
522 togglePin(watchDogPin);
523 }
524
525 // Tariff transitions must be tracked every second, whatever the override state:
526 // dual tariff forcing depends on the off-peak start timestamp set here.
527 if constexpr (DUAL_TARIFF)
528 {
529 bOffPeak = proceedDualTariffLogic();
530 }
531
532 // Router OFF: no forcing and no diversion, so every load and relay is OFF
533 const bool bRouterOff{ isRouterOff() };
534 const bool bDiversionEnabled{ !bRouterOff && isDiversionEnabled() };
535
536 // Get complete override bitmasks atomically (external pins + dual tariff forcing)
537 const OverrideMasks privateOverrideMasks{ bRouterOff ? OverrideMasks{ 0, 0 } : getOverrideBitmask(bOffPeak, iTemperature_x100) };
538
539 if constexpr (RELAY_DIVERSION)
540 {
541 relays.inc_duration();
542 relays.proceed_relays(privateOverrideMasks.local, bDiversionEnabled);
543 }
544
545 // Copy to the shared versions for ISR access
546 Shared::b_diversionEnabled = bDiversionEnabled;
547 Shared::overrideBitmask = privateOverrideMasks.local;
548 Shared::remoteOverrideBitmask = privateOverrideMasks.remote;
549
550 // Only process priority logic if no override pins are active (local or remote)
551 if (!Shared::overrideBitmask && !Shared::remoteOverrideBitmask)
552 {
553 proceedLoadPriorities();
554 }
555}

572void loop()
573{
574 static uint8_t perSecondTimer{ 0 };
575 static bool bOffPeak{ false };
576 static int16_t iTemperature_x100{ 0 };
577
578 // Write the datalog output in progress, as far as the serial buffer allows (never waits)
579 SerialOutput::poll();
580
581 // Process any pending RF transmissions (called outside ISR to avoid blocking)
582 if constexpr (REMOTE_LOADS_PRESENT)
583 {
584 processRemoteLoadTransmissions();
585 }
586
587 if (Shared::b_newMainsCycle) // flag is set after every pair of ADC conversions
588 {
589 Shared::b_newMainsCycle = false; // reset the flag
590 ++perSecondTimer;
591
592 if (perSecondTimer >= SUPPLY_FREQUENCY)
593 {
594 perSecondTimer = 0;
595 handlePerSecondTasks(bOffPeak, iTemperature_x100);
596 }
597 }
598
599 if (Shared::b_datalogEventPending)
600 {
601 Shared::b_datalogEventPending = false;
602
603 updatePowerAndVoltageData();
604
605 if constexpr (RELAY_DIVERSION)
606 {
607 relays.update_average(tx_data.power);
608 }
609
610 if constexpr (TEMP_SENSOR_PRESENT)
611 {
612 processTemperatureData();
613
614 // the first sensor is the reference for the dual tariff temperature threshold
615 iTemperature_x100 = tx_data.temperature_x100[0];
616 }
617
618 sendResults(bOffPeak);
619 }
620} // end of loop()
