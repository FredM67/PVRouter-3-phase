385void updatePowerAndVoltageData()
386{
387 // the calibration constants in fixed point, converted at compile time: no float code (#120)
388 static constexpr auto powerCal PROGMEM{ Datalog::powerCalibration(f_powerCal) };
389 static constexpr auto voltageCal PROGMEM{ Datalog::voltageCalibration(Datalog::vrmsFactor(DATALOG_PERIOD_IN_SECONDS), f_voltageCal) };
390
391 tx_data.power = 0;
392 uint8_t phase{ NO_OF_PHASES };
393 do
394 {
395 --phase;
396 const uint16_t sampleSets{ Shared::copyOf_sampleSetsDuringThisDatalogPeriod };
397
398 // import = +ve: the sum of the ISR is negative when importing
399 tx_data.power_L[phase] = Datalog::watts(Datalog::averagePower< DATALOG_POWER_SHIFT >(-Shared::copyOf_sumP_atSupplyPoint[phase], sampleSets),
400 pgm_read_word(&powerCal.value[phase]), powerCal.shift);
401
402 tx_data.power += tx_data.power_L[phase];
403
404 tx_data.Vrms_L_x100[phase] = Datalog::vrms(Shared::copyOf_sum_Vsquared[phase] / sampleSets,
405 pgm_read_word(&voltageCal.value[phase]), voltageCal.shift);
406 } while (phase);
407}

422void runTemperatureTransaction(const bool onTime)
423{
424 const TempSchedule::Action action{ tempSchedule.next(onTime) };
425 switch (action.step)
426 {
427 case TempSchedule::Step::CONVERT:
428 tempSensors.requestTemperatures();
429 break;
430 case TempSchedule::Step::ADDRESS:
431 if (!tempSensors.startRead(action.sensor))
432 {
433 temperatureReadings[action.sensor] = DEVICE_DISCONNECTED_RAW; // no answer to the reset
434 tempSchedule.skipRestOf(action.sensor);
435 }
436 break;
437 case TempSchedule::Step::READ:
438 tempSensors.readScratchpad(action.sensor);
439 break;
440 case TempSchedule::Step::CLOSE:
441 temperatureReadings[action.sensor] = tempSensors.endRead();
442 break;
443 default:
444 break;
445 }
446}

468void processTemperatureData(const bool bStartNow)
469{
470 // 85.00 °C more than 5.00 °C away from the last valid reading: the power-on value, skipped
471 static DS18B20::PowerOnFilter powerOnFilter[tempSensors.size()];
472
473 uint8_t idx{ tempSensors.size() };
474 do
475 {
476 --idx;
477 tx_data.temperature_x100[idx] = powerOnFilter[idx](temperatureReadings[idx]);
478 } while (idx);
479
480 // the next round: its conversion now, in the cycle that ends the period (which had no transaction,
481 // the round being over), or with the next cycle
482 if (tempSchedule.newPeriod() && bStartNow)
483 {
484 runTemperatureTransaction(true);
485 }
486}

512void handlePerSecondTasks(bool &bOffPeak, int16_t &iTemperature_x100)
513{
514 if constexpr (WATCHDOG_PIN_PRESENT)
515 {
516 togglePin(watchDogPin);
517 }
518
519 // Tariff transitions must be tracked every second, whatever the override state:
520 // dual tariff forcing depends on the off-peak start timestamp set here.
521 if constexpr (DUAL_TARIFF)
522 {
523 bOffPeak = proceedDualTariffLogic();
524 }
525
526 // Router OFF: no forcing and no diversion, so every load and relay is OFF
527 const bool bRouterOff{ isRouterOff() };
528 const bool bDiversionEnabled{ !bRouterOff && isDiversionEnabled() };
529
530 // Get complete override bitmasks atomically (external pins + dual tariff forcing)
531 const OverrideMasks privateOverrideMasks{ bRouterOff ? OverrideMasks{ 0, 0 } : getOverrideBitmask(bOffPeak, iTemperature_x100) };
532
533 if constexpr (RELAY_DIVERSION)
534 {
535 relayDiverter.inc_duration();
536 relayDiverter.proceed_relays(privateOverrideMasks.local, bDiversionEnabled);
537 }
538
539 // Copy to the shared versions for ISR access
540 Shared::b_diversionEnabled = bDiversionEnabled;
541 Shared::overrideBitmask = privateOverrideMasks.local;
542 Shared::remoteOverrideBitmask = privateOverrideMasks.remote;
543
544 // no rotation while anything is forced (local or remote)
545 proceedLoadPriorities(privateOverrideMasks.local || privateOverrideMasks.remote);
546}

564void loop()
565{
566 static uint8_t perSecondTimer{ 0 };
567 static bool bOffPeak{ false };
568 static int16_t iTemperature_x100{ 0 };
569
570 const bool bNewMainsCycle{ Shared::b_newMainsCycle }; // set once per mains cycle, by the ADC interrupt
571 if (bNewMainsCycle)
572 {
573 Shared::b_newMainsCycle = false; // reset the flag
574 }
575
576 [[maybe_unused]] uint32_t cycleSeen_us{ 0 }; // temperature sensing: when this pass saw the new cycle
577 [[maybe_unused]] bool bCycleOnTime{ false }; // ... and whether on time
578
579 if constexpr (TEMP_SENSOR_PRESENT)
580 {
581 // The flag was set at most one loop pass ago: on time if the last pass was short (the loop
582 // idle when the cycle began), late after a long one (per-second work, datalog, RF frame).
583 constexpr uint32_t ON_TIME_US{ 1000 };
584 static uint32_t passStart_us{ 0 };
585 const uint32_t now_us{ micros() };
586 const bool bOnTime{ (now_us - passStart_us) < ON_TIME_US };
587 passStart_us = now_us;
588
589 if (bNewMainsCycle)
590 {
591 cycleSeen_us = now_us;
592 bCycleOnTime = bOnTime;
593 runTemperatureTransaction(bOnTime); // first: its long slots must end before the next cycle's work
594 }
595 }
596
597 // Write the datalog output in progress, as far as the serial buffer allows (never waits)
598 SerialOutput::poll();
599
600 // Process any pending RF transmissions (called outside ISR to avoid blocking)
601 if constexpr (REMOTE_LOADS_PRESENT)
602 {
603 processRemoteLoadTransmissions();
604 }
605
606 if (bNewMainsCycle)
607 {
608 ++perSecondTimer;
609
610 if (perSecondTimer >= SUPPLY_FREQUENCY)
611 {
612 perSecondTimer = 0;
613 handlePerSecondTasks(bOffPeak, iTemperature_x100);
614 }
615 }
616
617 // With temperature sensing, the end of the period waits for the pass that sees the new cycle, so
618 // that the next round's conversion starts after the cycle's work and before the datalog output.
619 // The timing, a sample set being 6 conversions (624 µs, whatever the mains frequency):
620 // - L1's positive crossing: the period's datalog copies (processDataLogging(), from
621 // processPlusHalfCycle()), then b_datalogEventPending;
622 // - 2 sample sets later (1248 µs): the cycle's work, processStartNewCycle(), the heaviest call
623 // (its ADMUX written ~660 to 780 cycles in with 4 loads: sim/README.md, Temperature with remote
624 // loads, RF datalog and relays), then b_newMainsCycle;
625 // - the next such call, a mains cycle later.
626 // The conversion (reset, Skip ROM, Convert T: ~2.1 ms of bus I/O) cannot fit between the event
627 // and processStartNewCycle(); started after it, its slots fall among the ordinary calls.
628 if (Shared::b_datalogEventPending && (bNewMainsCycle || !TEMP_SENSOR_PRESENT))
629 {
630 Shared::b_datalogEventPending = false;
631
632 updatePowerAndVoltageData();
633
634 if constexpr (RELAY_DIVERSION)
635 {
636 relayDiverter.update_average(tx_data.power);
637 }
638
639 if constexpr (TEMP_SENSOR_PRESENT)
640 {
641 // On time if this pass saw the new cycle on time and is still within 3 ms of it (the per-second
642 // work may run in between): the conversion's slots then end within ~6 ms of the cycle's work
643 constexpr uint32_t PERIOD_END_ON_TIME_US{ 3000 };
644 processTemperatureData(bCycleOnTime && (micros() - cycleSeen_us) < PERIOD_END_ON_TIME_US);
645
646 // the sensor checked against the dual tariff temperature threshold
647 iTemperature_x100 = tx_data.temperature_x100[BOOST_TEMP_SENSOR];
648 }
649
650 sendResults(bOffPeak);
651 }
652} // end of loop()
