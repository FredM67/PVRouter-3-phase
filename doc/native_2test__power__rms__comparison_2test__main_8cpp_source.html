338uint16_t generateSineADC(uint16_t sampleIndex, uint16_t dcLevel, int16_t amplitude, float phaseShift = 0.0f)
339{
340 float angle = 2.0f * PI * sampleIndex / SAMPLES_PER_CYCLE + phaseShift;
341 int16_t ac = static_cast< int16_t >(amplitude * sinf(angle));
342 int16_t adc = dcLevel + ac;
343 if (adc < 0) adc = 0;
344 if (adc > 1023) adc = 1023;
345 return static_cast< uint16_t >(adc);
