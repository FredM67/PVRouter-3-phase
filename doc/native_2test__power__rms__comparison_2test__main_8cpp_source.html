45

52

60 {

62 }
63

68 {
69 // Remove DC offset (using nominal mid-point)
70 int32_t sampleIminusDC = (static_cast< int32_t >(rawSample - 512)) << 8;
71
72 // Calculate power: reduce to 16-bits, multiply, scale back
73 const int32_t filtV_div4 = l_sampleVminusDC >> 2; // x64

76 instP >>= 12; // x1

85 void processVoltage()

92 l_cumVdeltasThisCycle += l_sampleVminusDC;

261
262public:

264 : state(seed) {}

266 void seed(uint32_t s)
267 {
268 state = s ? s : 1;
269 }
270
271 uint32_t next()
272 {
273 state ^= state << 13;
274 state ^= state >> 17;
275 state ^= state << 5;
276 return state;
277 }
278
282 int16_t noise(int16_t amplitude)
283 {
284 if (amplitude == 0) return 0;
285 return static_cast< int16_t >((next() % (2 * amplitude + 1)) - amplitude);
286 }

339uint16_t generateSineADC(uint16_t sampleIndex, uint16_t dcLevel, int16_t amplitude, float phaseShift = 0.0f)
340{
341 float angle = 2.0f * PI * sampleIndex / SAMPLES_PER_CYCLE + phaseShift;
342 int16_t ac = static_cast< int16_t >(amplitude * sinf(angle));
343 int16_t adc = dcLevel + ac;
344 if (adc < 0) adc = 0;
345 if (adc > 1023) adc = 1023;
346 return static_cast< uint16_t >(adc);
347}

352uint16_t generateSineADCWithNoise(uint16_t sampleIndex, uint16_t dcLevel, int16_t amplitude,
353 int16_t noiseAmplitude, float phaseShift = 0.0f)
354{
355 float angle = 2.0f * PI * sampleIndex / SAMPLES_PER_CYCLE + phaseShift;
356 int16_t ac = static_cast< int16_t >(amplitude * sinf(angle));
357 int16_t noise = rng.noise(noiseAmplitude);
358 int16_t adc = dcLevel + ac + noise;
359 if (adc < 0) adc = 0;
360 if (adc > 1023) adc = 1023;
361 return static_cast< uint16_t >(adc);
362}

402 struct TestCase

405 uint16_t current_adc;
406 };
407
408 TestCase cases[] = {
409 { 512, 512 }, // Both at mid-point (zero power)
410 { 612, 562 }, // Positive V, positive I
411 { 412, 462 }, // Negative V, negative I
412 { 612, 462 }, // Positive V, negative I
413 { 412, 562 }, // Negative V, positive I
414 { 912, 712 }, // High values
415 { 112, 312 }, // Low values
416 };
417
418 for (const auto& tc : cases)
419 {
420 // Create fresh implementations with MATCHING DC offset (512)
421 OldImplementation oldTest;
422 NewImplementation newTest;

432 newTest.processCurrentRawSample(toLeftAligned(tc.current_adc));
433
434 // With matching DC, difference is only from rounding (~1.5% per sample)
435 // This averages out to near-zero over a full cycle
