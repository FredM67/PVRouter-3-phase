118void test_each_pin_keeps_its_own_level(void)

136void test_press_release_press(void)
137{

169void test_a_press_while_forced_is_dropped_not_carried_over(void)
170{
171 // the rotation pin is read every second, forced or not
172 TEST_ASSERT_FALSE(isRotationRequested(trackPinEdges< 12 >(HIGH_), false));
173
174 // pressed during forcing: no rotation
175 TEST_ASSERT_FALSE(isRotationRequested(trackPinEdges< 12 >(LOW_), true));
176 TEST_ASSERT_FALSE(isRotationRequested(trackPinEdges< 12 >(LOW_), true));
177
178 // forcing ends, the pin still held: no edge any more, no rotation
179 TEST_ASSERT_FALSE(isRotationRequested(trackPinEdges< 12 >(LOW_), false));
180 TEST_ASSERT_FALSE(isRotationRequested(trackPinEdges< 12 >(HIGH_), false));
181
182 // a new press after forcing rotates
183 TEST_ASSERT_TRUE(isRotationRequested(trackPinEdges< 12 >(LOW_), false));
184}
