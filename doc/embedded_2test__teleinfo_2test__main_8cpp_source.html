31/* one line as the router writes it: LF tag TAB value TAB checksum CR */
32static void appendLine(char* expected, const char* tagAndValue)
33{
34 uint8_t sum{ 0 };
35 for (const char* p = tagAndValue; *p; ++p)
36 {
37 sum += *p;
38 }
39 sum += '\t';
40 const char line[]{ '\n', '\0' };
41 strcat(expected, line);
42 strcat(expected, tagAndValue);
43 const char end[]{ '\t', static_cast< char >((sum & 0x3F) + 0x20), '\r', '\0' };
44 strcat(expected, end);
45}

56void test_lineSize_calculation(void)
57{
58 // LF + tag + TAB + value + TAB + checksum + CR = 5 + tagLength + valueLength
59 TEST_ASSERT_EQUAL(5 + 1 + 3, TeleInfo::lineSize(1, 3));

153/* captures a frame, written step by step as the serial output does */
154class FramePrint : public Print
155{
156public:
157 char text[240]{};
158 size_t length{ 0 };
159
160 using Print::write;
161
162 size_t write(uint8_t c) override
163 {
164 if (length < sizeof text - 1)
165 {
166 text[length++] = static_cast< char >(c);
167 }
168 return 1;
169 }
170};
171
172template< typename Source >
173static void streamFrame(FramePrint& out)
174{
175 uint8_t step{ 0 };
176 bool more{ true };
177 while (more)
178 {
179 char text[TeleInfo::MAX_STEP_LENGTH];
180 uint8_t length{ 0 };
181 more = TeleInfo::writeFrameStep< Source >(step++, text, length);
182 TEST_ASSERT_LESS_OR_EQUAL(TeleInfo::MAX_STEP_LENGTH, length);
183 out.write(text, length);
184 }
185}
