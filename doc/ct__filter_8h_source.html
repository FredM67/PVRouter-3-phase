72constexpr Coefficients coefficientsOf(const float gain, const float alpha)
73{
74 uint8_t shift{ MAX_ALPHA_SHIFT };
75 while ((shift != 0) && (alpha * static_cast< float >(1UL << (16 + shift)) + 0.5F >= static_cast< float >(ALPHA_FIXED_LIMIT)))

77 --shift;
78 }

80 return { alphaFixed,

83 shift,

95constexpr bool isAccurate(const float gain, const float alpha)
96{
97 if (!(alpha > 0.0F) || !(alpha < 0.0625F) || !(gain >= 0.0F) || !(gain < 64.0F))
98 {
99 return false;
100 }
101 const Coefficients coefficients{ coefficientsOf(gain, alpha) };
102 const float fixedAlpha{ static_cast< float >(coefficients.alpha) / static_cast< float >(1UL << (16 + coefficients.alphaShift)) };
103 const float fixedGain{ static_cast< float >(coefficients.gain) / 256.0F };
104 const float alphaError{ fixedAlpha > alpha ? fixedAlpha - alpha : alpha - fixedAlpha };
105 const float gainError{ fixedGain > gain ? fixedGain - gain : gain - fixedGain };
106 return (coefficients.alpha < ALPHA_FIXED_LIMIT) && (alphaError <= alpha * 0.01F) && (gainError <= gain * 0.01F);
107}

137 int16_t apply(const int16_t sampleMinusDC, const Coefficients &coefficients)
138 {
139 // alpha x (4 x sample - lpf), in units of 2^-(16 + shift) at x256:
140 // 4 x alpha x sample - 4 x alpha x lpfX64 - alpha x lpfLowBits
141 int32_t alphaSample;
142 int32_t alphaLpf;
143 multS16x16_to32(alphaSample, sampleMinusDC, coefficients.alphaX4);
144 multS16x16_to32(alphaLpf, lpfX64, coefficients.alphaX4);
145 int32_t increment{ alphaSample - alphaLpf }; // below 2^30
146 if (lpfLowBits & 1)
147 {
148 increment -= coefficients.alpha;
149 }
150 if (lpfLowBits & 2)
151 {
152 increment -= coefficients.alphaX2;
153 }
154
155 // lpf + increment / 2^(16 + shift), truncated towards zero: floored, plus one when the
156 // result is negative with a fraction dropped
157 const uint8_t shift{ coefficients.alphaShift };
158 const int16_t incrementHigh{ static_cast< int16_t >(increment >> 16) }; // a word move, below 2^14
159 const bool fractionDropped{ (static_cast< uint16_t >(increment) != 0) || ((incrementHigh & ((1 << shift) - 1)) != 0) };
160 const int16_t sum{ static_cast< int16_t >(lpfLowBits + (incrementHigh >> shift)) };
161 lpfX64 += sum >> 2; // floored
162 lpfLowBits = static_cast< uint8_t >(sum & 3);
163 if ((lpfX64 < 0) && fractionDropped)
164 {
165 if (++lpfLowBits == 4)
166 {
167 lpfLowBits = 0;
168 ++lpfX64;
169 }
170 }
171
172 // gain x lpf at x64, x 2^8: gain x 2^8 x lpfX64 + gain x 2^6 x lpfLowBits
173 int32_t boost;
174 multS16x16_to32(boost, lpfX64, coefficients.gain); // below 2^29: the gain below 64
175 if (lpfLowBits & 1)
176 {
177 boost += coefficients.gainQuarter;
178 }
179 if (lpfLowBits & 2)
180 {
181 boost += coefficients.gainHalf;
182 }
183
184 const int32_t sample{ static_cast< int32_t >(sampleMinusDC) + (boost >> 8) }; // a byte move
185
186 if (sample > INT16_MAX)
187 {
188 return INT16_MAX;
189 }
190 if (sample < INT16_MIN)
191 {
192 return INT16_MIN;
193 }
194 return static_cast< int16_t >(sample);
195 }
