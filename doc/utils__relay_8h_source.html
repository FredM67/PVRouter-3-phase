167 bool proceed_relay(const int32_t currentAvgPower, const uint16_t overrideBitmask) const
168 {
169 // To avoid changing sign, surplus is a negative value
170 // Turn ON if surplus threshold is met OR if override is active
171 if (currentAvgPower < surplusThreshold || isForced(overrideBitmask))
172 {
173 return try_switch(true);
174 }
175
176 // Positive threshold: turn OFF when importing > threshold (normal mode).
177 // Negative threshold: turn OFF when surplus < abs(threshold) (battery mode).
178 if (currentAvgPower > importThreshold)
179 {
180 return try_switch(false);
181 }
182
183 return false;
184 }

384 void proceed_relays(const uint16_t overrideBitmask, const bool diversionEnabled = true) const
385 {
386 if (settle_change != 0)
387 {
388 // A relay has been toggle less than a minute ago, wait until changes take effect
389 return;
390 }
391
392 if (!diversionEnabled)
393 {
394 bool anyChanges{ false };
395 uint8_t idx{ N };
396 do
397 {
398 anyChanges |= relay[--idx].proceed_forced_only(overrideBitmask);
399 } while (idx);
400
401 if (anyChanges)
402 settle_change = 60;
403
404 return;
405 }
406
407 const auto currentAvgPower{ ewma_average.getAverageT() };
408
409 if (currentAvgPower > 0)
410 {
411 // Currently importing, try to turn OFF some relays
412 uint8_t idx{ N };
413 do
414 {
415 if (relay[--idx].proceed_relay(currentAvgPower, overrideBitmask))
416 {
417 settle_change = 60;
418 return;
419 }
420 } while (idx);
421 }
422 else
423 {
424 // Remaining surplus, try to turn ON more relays
425 uint8_t idx{ 0 };
426 do
427 {
428 if (relay[idx].proceed_relay(currentAvgPower, overrideBitmask))
429 {
430 settle_change = 60;
431 return;
432 }
433 } while (++idx < N);
434 }
435 }
