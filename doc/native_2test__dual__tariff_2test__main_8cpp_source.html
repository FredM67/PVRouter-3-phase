151void test_boost_window_across_the_millis_wraparound(void)
152{
153 const uint32_t offPeakStart{ UINT32_MAX - MINUTE + 1 }; // 1 min before millis() wraps
154 const uint32_t start{ DualTariff::windowStart(LAST(HOURS(3)), 8) }; // 5 h
155 const uint32_t end{ DualTariff::windowEnd(LAST(HOURS(3)), 8) };
156
157 const uint32_t at6h{ offPeakStart + 6 * HOUR }; // wrapped around
158 const uint32_t at4h{ offPeakStart + 4 * HOUR };
159
160 TEST_ASSERT_TRUE(at6h < offPeakStart);
161 TEST_ASSERT_TRUE(DualTariff::inWindow(at6h - offPeakStart, start, end));
162 TEST_ASSERT_FALSE(DualTariff::inWindow(at4h - offPeakStart, start, end));
