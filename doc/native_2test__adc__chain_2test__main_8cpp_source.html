33static Channel channelOfAdmux(const uint8_t admux)

126void test_pipeline_order(void)
127{
128 constexpr uint8_t ROUNDS{ 60 };
129
130 for (const PcbVersion board : boards)
131 {
132 // admuxOf[k]: the ADMUX the conversion of round k runs with
133 uint8_t admuxOf[ROUNDS + 2]{};
134
135 // the chain primed with V1, then I1
136 admuxOf[0] = AdcChain::admuxOf(board, Channel::V1);
137 admuxOf[1] = AdcChain::admuxOf(board, Channel::I1);
138
139 for (uint8_t round = 0; round != ROUNDS; ++round)
140 {
141 // the ISR of round k believes it handles channelConverted(k)...
142 const Channel believed{ AdcChain::channelConverted(round) };
143
144 // ... and the ADC really converted the channel its ADMUX selected
145 TEST_ASSERT_EQUAL_UINT8(static_cast< uint8_t >(believed), static_cast< uint8_t >(channelOfAdmux(admuxOf[round])));
146 TEST_ASSERT_EQUAL_HEX8(AdcChain::refBits(board), admuxOf[round] & REF_MASK);
147 TEST_ASSERT_EQUAL_HEX8(ADLAR_BIT, admuxOf[round] & ADLAR_BIT);
148
149 // the conversion of round k+1 has already started: the write of round k applies to round k+2
150 admuxOf[round + 2] = AdcChain::admuxWrittenAfter(board, believed);
151 }
152 }
153}

155void test_channel_table_matches_the_former_literals(void)
156{
157 // the table processing.cpp held by hand: { next, index, admux }, ADMUX = reference | ADLAR | pin
158 struct Former
159 {
160 uint8_t index;
161 uint8_t setupPin;
162 };
163 constexpr Former former[6]{
164 { 0, 2 }, // V1 -> setup for V2
165 { 1, 3 }, // I1 -> setup for I2
166 { 2, 4 }, // V2 -> setup for V3
167 { 3, 5 }, // I2 -> setup for I3
168 { 4, 0 }, // V3 -> setup for V1
169 { 5, 1 } // I3 -> setup for I1
170 };
171
172 for (const PcbVersion board : boards)
173 {
174 const uint8_t ref{ board == PcbVersion::NEW ? static_cast< uint8_t >(INTERNAL_1V1) : AVCC };
175 for (const Channel ch : chain)
176 {
177 const Former &f{ former[static_cast< uint8_t >(ch)] };
178 TEST_ASSERT_EQUAL_UINT8(f.index, static_cast< uint8_t >(ch));
179 TEST_ASSERT_EQUAL_HEX8(ref | ADLAR_BIT | f.setupPin, AdcChain::admuxWrittenAfter(board, ch));
180 }
181 // the start-up ADMUX: reference, ADLAR, V1 (_ADMUX | sensorV[0])
182 TEST_ASSERT_EQUAL_HEX8(ref | ADLAR_BIT | 0, AdcChain::admuxOf(board, Channel::V1));
183 }
