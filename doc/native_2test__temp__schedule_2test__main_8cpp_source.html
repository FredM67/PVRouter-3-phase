65void test_round_of_two_sensors()
66{
67 TempSchedule::Scheduler< 2, C50 > schedule;
68
69 assertAction({ Step::CONVERT, 0 }, schedule.next(true), "first cycle: the conversion");
70 for (uint16_t i = 1; i != C50; ++i)
71 {
72 assertAction({ Step::NONE, 0 }, schedule.next(true), "during the conversion: nothing");
73 }
74 for (uint16_t k = 0; k != 6; ++k)
75 {
76 assertAction(readStep(k), schedule.next(true), "the reads, one transaction per cycle");
77 }
78 TEST_ASSERT_TRUE(schedule.done());
79 for (uint16_t i = 0; i != 300; ++i)
80 {
81 assertAction({ Step::NONE, 0 }, schedule.next(true), "round over: nothing until the next period");
82 }

168void test_conversion_in_the_cycle_that_ends_the_period()
169{
170 // The loop starts the next round as soon as it handles the end of the period, in the same mains
171 // cycle, if it is on time: the round's cycle had no transaction (round over), so that cycle
172 // still has one at most, and the bus is quiet again before the datalog output starts.
173 TempSchedule::Scheduler< 1, C50 > schedule;
174 for (uint16_t i = 0; i != C50 + 3; ++i)
175 {
176 schedule.next(true);
177 }
178 assertAction({ Step::NONE, 0 }, schedule.next(true), "the cycle that ends the period: nothing");
179 TEST_ASSERT_TRUE(schedule.newPeriod());
180 assertAction({ Step::CONVERT, 0 }, schedule.next(true), "the conversion, in that same cycle");
181 for (uint16_t i = 1; i != C50; ++i)
182 {
183 assertAction({ Step::NONE, 0 }, schedule.next(true), "during the conversion: nothing");
184 }
185 assertAction({ Step::ADDRESS, 0 }, schedule.next(true), "");
186}

201void test_never_two_transactions_in_one_cycle_and_never_before_the_conversion_time()
202{
203 // a pseudo-random pattern of late cycles, over many periods
204 TempSchedule::Scheduler< 4, C50 > schedule;
205 uint32_t lfsr{ 0xACE1U };
206 int32_t lastConvert{ -100000 };
207 uint16_t rounds{ 0 };
208 for (int32_t cycle = 0; cycle != 20000; ++cycle)
209 {
210 lfsr = (lfsr >> 1) ^ (-(lfsr & 1U) & 0xB400U);
211 const bool onTime{ (lfsr & 7U) != 0 }; // 1 cycle in 8 late
212 if (cycle % 250 == 0)
213 {
214 schedule.newPeriod();
215 }
216 const Action a{ schedule.next(onTime) }; // one call per cycle: one transaction at most
217 if (a.step != Step::NONE)
218 {
219 TEST_ASSERT_TRUE_MESSAGE(onTime, "a transaction on a late cycle");
220 }
221 if (a.step == Step::CONVERT)
222 {
223 lastConvert = cycle;
224 ++rounds;
225 }
226 if (a.step == Step::ADDRESS || a.step == Step::READ)
227 {
228 TEST_ASSERT_TRUE_MESSAGE(cycle - lastConvert >= C50, "a read before the end of the conversion");
229 }
230 }
231 TEST_ASSERT_EQUAL_UINT16(80, rounds); // one round per period
232}

245template< uint8_t N, uint8_t F >
246static bool roundEndsInPeriod(const uint16_t periodCycles)
247{
248 constexpr uint16_t C{ TempSchedule::conversionCycles(F) };
249 TempSchedule::Scheduler< N, C > schedule;
250 schedule.newPeriod();
251 for (uint16_t tick = 1; tick != periodCycles; ++tick) // tick periodCycles: the period-end cycle
252 {
253 const bool onTime{ (tick % F) != 1 };
254 schedule.next(onTime);
255 if (schedule.done())
256 {
257 return true;
258 }
259 }
260 return false;
261}

263void test_rounds_fit_their_period()
264{
265 // 5 s, the default
266 static_assert(TempSchedule::fitsInPeriod(2, 50, 250), "");
267 static_assert(TempSchedule::fitsInPeriod(8, 60, 300), "");
268 TEST_ASSERT_TRUE((roundEndsInPeriod< 2, 50 >(250)));
269 TEST_ASSERT_TRUE((roundEndsInPeriod< 8, 60 >(300)));
270 // 1 s: the conversion takes most of it
271 static_assert(TempSchedule::fitsInPeriod(3, 50, 50), "");
272 static_assert(!TempSchedule::fitsInPeriod(4, 50, 50), "");
273 static_assert(TempSchedule::fitsInPeriod(4, 60, 60), "");
274 static_assert(!TempSchedule::fitsInPeriod(5, 60, 60), "");
275 TEST_ASSERT_TRUE((roundEndsInPeriod< 3, 50 >(50)));
276 TEST_ASSERT_TRUE((roundEndsInPeriod< 4, 60 >(60)));
277 // what fitsInPeriod() refuses does not fit
278 TEST_ASSERT_FALSE((roundEndsInPeriod< 5, 50 >(50)));
279 // 2 s: the late cycle within the round counts
280 static_assert(TempSchedule::fitsInPeriod(19, 50, 100), "");
281 static_assert(!TempSchedule::fitsInPeriod(20, 50, 100), "");
282 TEST_ASSERT_TRUE((roundEndsInPeriod< 19, 50 >(100)));
283 TEST_ASSERT_FALSE((roundEndsInPeriod< 20, 50 >(100)));
284}
