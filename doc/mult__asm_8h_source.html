68 "clr r1 \n\t" // Restore r1 to zero

142template< typename A, typename B >
143static inline __attribute__((always_inline)) void multU24x16_to32_hi8(uint32_t& result, A a, B b)
144{
145 static_assert(is_same_v< A, uint32_t >, "First argument must be uint32_t (24 bits used)");
146 static_assert(is_same_v< B, uint16_t >, "Second argument must be uint16_t");
147
148#ifdef __AVR__
149 // Byte k of the result is byte k+1 of the product; product byte 0 is dropped.
150 asm volatile(
151 "clr r26 \n\t" // Clear temporary register
152 "mul %A1, %A2 \n\t" // a0 * b0 -> product bytes 0-1
153 "mov %A0, r1 \n\t" // keep byte 1 only
154 "clr %B0 \n\t"
155 "clr %C0 \n\t"
156 "clr %D0 \n\t"
157 "mul %A1, %B2 \n\t" // a0 * b1 -> product bytes 1-2
158 "add %A0, r0 \n\t"
159 "adc %B0, r1 \n\t"
160 "adc %C0, r26 \n\t"
161 "mul %B1, %A2 \n\t" // a1 * b0 -> product bytes 1-2
162 "add %A0, r0 \n\t"
163 "adc %B0, r1 \n\t"
164 "adc %C0, r26 \n\t"
165 "mul %B1, %B2 \n\t" // a1 * b1 -> product bytes 2-3
166 "add %B0, r0 \n\t"
167 "adc %C0, r1 \n\t"
168 "adc %D0, r26 \n\t"
169 "mul %C1, %A2 \n\t" // a2 * b0 -> product bytes 2-3
170 "add %B0, r0 \n\t"
171 "adc %C0, r1 \n\t"
172 "adc %D0, r26 \n\t"
173 "mul %C1, %B2 \n\t" // a2 * b1 -> product bytes 3-4
174 "add %C0, r0 \n\t"
175 "adc %D0, r1 \n\t" // no carry out: a < 2^24, b < 2^16
176 "clr r1 \n\t" // Restore r1 to zero
177 : "=&r"(result)
178 : "r"(a), "r"(b)
179 : "r26");
180#else
181 result = static_cast< uint32_t >((static_cast< uint64_t >(a) * b) >> 8);
182#endif
183}

204template< typename V, typename F >
205static inline __attribute__((always_inline)) void mult16x8_q8(int16_t& result, V value, F fraction)
206{
207 static_assert(is_same_v< V, int16_t >, "Value argument must be int16_t");
208 static_assert(is_same_v< F, uint8_t >, "Fraction argument must be uint8_t");
209
210#ifdef __AVR__
211 asm volatile(
212 "mulsu %B[val], %[frac] \n\t" // value_hi * fraction (signed*unsigned)
213 "movw %A[ret], r0 \n\t" // Store result
214 "mul %A[val], %[frac] \n\t" // value_lo * fraction
215 "lsl r0 \n\t" // Shift for rounding
216 "adc %A[ret], r1 \n\t" // Add with carry (rounding)
217 "clr r1 \n\t" // Restore r1 to zero
218 "adc %B[ret], r1 \n\t" // Propagate carry
219 : [ret] "=&r"(result)
220 : [val] "a"(value), [frac] "a"(fraction));
221#else
222 result = static_cast< int16_t >((static_cast< int32_t >(value) * fraction + 0x80) >> 8);
223#endif
224}
