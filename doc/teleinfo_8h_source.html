100inline char* writeLine(char* dst, const char* tag, uint8_t index, uint16_t magnitude, bool negative)
101{
102 *dst++ = LF;
103
104 char* const start{ dst };
105 while (*tag)
106 {
107 *dst++ = *tag++;
108 }
109 if (index != 0)
110 {
111 *dst++ = static_cast< char >('0' + index);

168{

172 {
173 steps += 2 * Source::PHASE_COUNT; // Pn, Vn ... P1, V1

198template< typename Source >
199char* frameLine(uint8_t step, char* dst)
200{
201 if (step == 0)
202 {
203 return line(dst, "P", Source::power());
204 }
205 --step;
206
207 if constexpr (Source::PHASE_COUNT > 1)
208 {
209 if (step < 2 * Source::PHASE_COUNT) // Pn, Vn first
210 {
211 const uint8_t phase{ static_cast< uint8_t >(Source::PHASE_COUNT - 1 - step / 2) };
212 return (step & 1) == 0 ? line(dst, "P", Source::phasePower(phase), phase + 1) : lineUnsigned(dst, "V", Source::phaseVoltage(phase), phase + 1);
213 }
214 step -= 2 * Source::PHASE_COUNT;
215 }
216
217 if constexpr (Source::RELAY_DIVERSION)
218 {
219 if (step == 0)
220 {
221 return line(dst, "R", Source::relayAverage());
222 }
223 --step;
224
225 if (step < Source::RELAY_COUNT) // Rn first
226 {
227 const uint8_t index{ static_cast< uint8_t >(Source::RELAY_COUNT - step) };
228 return line(dst, "R", Source::relayOn(index - 1), index);
229 }
230 step -= Source::RELAY_COUNT;
231 }
232
233 if (step < Source::LOAD_COUNT) // Dn first
234 {
235 const uint8_t index{ static_cast< uint8_t >(Source::LOAD_COUNT - step) };
236 return line(dst, "D", Source::loadOnPercent(index - 1), index);
237 }
238 step -= Source::LOAD_COUNT;
239
240 if constexpr (Source::TEMP_COUNT != 0)
241 {
242 if (step < Source::TEMP_COUNT)
243 {
244 return Source::temperatureValid(step) ? line(dst, "T", Source::temperature(step), step + 1) : dst;
245 }
246 step -= Source::TEMP_COUNT;
247 }
248
249 if (step == 0)
250 {
251 return lineUnsigned(dst, "N", Source::absenceCount());
252 }
253 --step;
254
255 if constexpr (Source::DUAL_TARIFF)
256 {
257 if (step == 0)
258 {
259 return line(dst, "TA", Source::offPeak() ? 1 : 0);
260 }
261 --step;
262 }
263
264 if (step == 0)
265 {
266 return lineUnsigned(dst, "S", Source::sampleSets());
267 }
268 return line(dst, "S_MC", Source::lowestSampleSetsPerCycle());
269}
