113template< uint8_t NMin, uint8_t NMax, uint8_t N >
114constexpr PerSampleCalibration< NMin, NMax, N > toFixedPerSample(const float (&cal)[N])
115{
116 static_assert((NMin >= 1) && (NMin <= NMax), "empty range of sample counts");
117 static_assert(NMax <= 64, "sumP must stay within 24 bits: at most 64 sample sets per cycle");
118
119 PerSampleCalibration< NMin, NMax, N > table{};
120 table.shift = shiftFor(largestOf(cal) / NMin); // the largest entry: largest cal, smallest n
121 for (uint8_t phase = 0; phase < N; ++phase)
122 {
123 for (uint8_t n = NMin; n <= NMax; ++n)
124 {
125 table.value[phase][n - NMin] = toFixed(cal[phase] / n, table.shift);
126 }
127 }
128 return table;
129}

161inline int32_t contribution(const int32_t averagePower, const uint16_t calFixed, const uint8_t shift)
162{
163 // Work on the magnitude and apply the sign last: rounding is then half away from
164 // zero, and import and export weigh exactly the same.
165 const bool negative{ averagePower < 0 };
166 const uint32_t magnitude{ static_cast< uint32_t >(negative ? -averagePower : averagePower) };
167
168 uint32_t product; // (magnitude x calFixed) >> 8
169 multU24x16_to32_hi8(product, magnitude, calFixed);
170
171 // remaining shift down to bucket units; >= 1 for any calibration value below 8
172 const uint8_t remaining{ static_cast< uint8_t >(shift - FRACTION_BITS - 8) };
173 const auto rounded{ static_cast< int32_t >((product + (1UL << (remaining - 1))) >> remaining) };
174
175 return negative ? -rounded : rounded;
176}

203template< uint8_t SUPPLY_FREQUENCY >
204constexpr int32_t toHundredthsOfJoule(const int32_t units)
205{
206 constexpr uint16_t unitsPerJoule{ SUPPLY_FREQUENCY << FRACTION_BITS };
207 constexpr int32_t numerator{ 100 / gcd(100, unitsPerJoule) };
208 constexpr int32_t denominator{ unitsPerJoule / gcd(100, unitsPerJoule) };
209
210 return (units * numerator + (units < 0 ? -denominator / 2 : denominator / 2)) / denominator;
211}
