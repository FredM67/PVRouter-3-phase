39{

114template< uint8_t NMin, uint8_t NMax, uint8_t N >
115constexpr PerSampleCalibration< NMin, NMax, N > toFixedPerSample(const float (&cal)[N])
116{
117 static_assert((NMin >= 1) && (NMin <= NMax), "empty range of sample counts");
118 static_assert(NMax <= 64, "sumP must stay within 24 bits: at most 64 sample sets per cycle");
119
120 PerSampleCalibration< NMin, NMax, N > table{};
121 table.shift = shiftFor(largestOf(cal) / NMin); // the largest entry: largest cal, smallest n
122 for (uint8_t phase = 0; phase < N; ++phase)
123 {
124 for (uint8_t n = NMin; n <= NMax; ++n)
125 {
126 table.value[phase][n - NMin] = toFixed(cal[phase] / n, table.shift);
127 }
128 }
129 return table;
130}

162inline int32_t contribution(const int32_t averagePower, const uint16_t calFixed, const uint8_t shift)
163{
164 // Work on the magnitude and apply the sign last: rounding is then half away from
165 // zero, and import and export weigh exactly the same.
166 const bool negative{ averagePower < 0 };
167 const uint32_t magnitude{ static_cast< uint32_t >(negative ? -averagePower : averagePower) };
168
169 uint32_t product; // (magnitude x calFixed) >> 8
170 multU24x16_to32_hi8(product, magnitude, calFixed);
171
172 // remaining shift down to bucket units; >= 1 for any calibration value below 8
173 const uint8_t remaining{ static_cast< uint8_t >(shift - FRACTION_BITS - 8) };
174
175 // Rounded: (product + 2^(remaining-1)) >> remaining. From remaining = 9 on, the rounding
176 // constant is a multiple of 256, so the same value is ((product >> 8) + 2^(remaining-9))
177 // >> (remaining-8): '>> 8' is a byte move, and the add in between keeps the compiler from
178 // merging both shifts back into one loop (remaining = 11 for the new board with 50 A CTs:
179 // a 3-turn loop instead of 11, ~52 cycles less). It also cannot overflow, unlike the
180 // 32-bit add of the general form. shift is a constant at the call site in the ISR, so
181 // only one branch is compiled there.
182 int32_t rounded;
183 if (remaining >= 9)
184 {
185 rounded = static_cast< int32_t >(((product >> 8) + (1UL << (remaining - 9))) >> (remaining - 8));
186 }
187 else
188 {
189 rounded = static_cast< int32_t >((product + (1UL << (remaining - 1))) >> remaining);
190 }
191
192 return negative ? -rounded : rounded;
193}

220template< uint8_t SUPPLY_FREQUENCY >
221constexpr int32_t toHundredthsOfJoule(const int32_t units)
222{
223 constexpr uint16_t unitsPerJoule{ SUPPLY_FREQUENCY << FRACTION_BITS };
224 constexpr int32_t numerator{ 100 / gcd(100, unitsPerJoule) };
225 constexpr int32_t denominator{ unitsPerJoule / gcd(100, unitsPerJoule) };
226
227 return (units * numerator + (units < 0 ? -denominator / 2 : denominator / 2)) / denominator;
228}
