69constexpr uint16_t check_pins()
70{
71 uint16_t used_pins{ 0 };
72
73 if constexpr (TEMP_SENSOR_PRESENT)
74 {
75 if (temperatureSensing.get_pin() != unused_pin)
76 bit_set(used_pins, temperatureSensing.get_pin());
77 }
78
79 if (diversionPin != unused_pin)
80 {
81 if (bit_read(used_pins, diversionPin))
82 return 0;
83
84 bit_set(used_pins, diversionPin);
85 }
86
87 if (routerOffPin != unused_pin)
88 {
89 if (bit_read(used_pins, routerOffPin))
90 return 0;
91
92 bit_set(used_pins, routerOffPin);
93 }
94
95 if (rotationPin != unused_pin)
96 {
97 if (bit_read(used_pins, rotationPin))
98 return 0;
99
100 bit_set(used_pins, rotationPin);
101 }
102
103 if (dualTariffPin != unused_pin)
104 {
105 if (bit_read(used_pins, dualTariffPin))
106 return 0;
107
108 bit_set(used_pins, dualTariffPin);
109 }
110
111 // Check override pins from the flexible override system
112 if constexpr (OVERRIDE_PIN_PRESENT)
113 {
114 for (uint8_t i = 0; i < overridePins.size(); ++i)
115 {
116 const uint8_t pin = overridePins.getPin(i);
117 if (pin != unused_pin)
118 {
119 if (bit_read(used_pins, pin))
120 return 0;
121
122 bit_set(used_pins, pin);
123 }
124 }
125 }
126
127 if (watchDogPin != unused_pin)
128 {
129 if (bit_read(used_pins, watchDogPin))
130 return 0;
131
132 bit_set(used_pins, watchDogPin);
133 }
134
135 // physicalLoadPin: the local TRIACs, plus the optional status LED of each remote load
136 for (const auto &loadEntry : physicalLoadPin)
137 {
138 const uint8_t pin{ Load::pinOf(loadEntry) };
139
140 // A remote load may legitimately have no status LED.
141 // Note that a local entry left at 'unused_pin' (0xFF) decodes as unit 3 / pin 63,
142 // which Load::isValidMap() rejects - the pin field alone cannot carry that sentinel.
143 if (!Load::isLocal(loadEntry) && (pin == 0))
144 continue;
145
146 // out of range for the bitmask - Load::isValidMap() reports what is actually wrong
147 if (pin > 15)
148 return 0;
149
150 if (bit_read(used_pins, pin))
151 return 0;
152
153 bit_set(used_pins, pin);
154 }
155
156 if constexpr (RELAY_DIVERSION)
157 {
158 for (uint8_t idx = 0; idx < relays.size(); ++idx)
159 {
160 const auto relayPin = relays.get_relay(idx).get_pin();
161
162 if (relayPin != unused_pin)
163 {
164 if (bit_read(used_pins, relayPin))
165 return 0;
166
167 bit_set(used_pins, relayPin);
168 }
169 }
170 }
171
172 return used_pins;
173}

175constexpr bool check_load_priorities()
176{
177 uint8_t _sum{ 0 };
178 uint16_t _enum_val{ 0 };
179
180 for (const auto &loadPrio : loadPrioritiesAtStartup)
181 {
182 // we check if a prio is used twice
183 if (bit_read(_enum_val, loadPrio))
184 return 0;
185
186 bit_set(_enum_val, loadPrio);
187
188 _sum += loadPrio;
189 }
190
191 // check if we have all prio between 0 and (NO_OF_DUMPLOADS - 1)
192 return _sum == ((NO_OF_DUMPLOADS * (NO_OF_DUMPLOADS - 1)) >> 1);
193}

199constexpr bool check_override_targets()
200{
201 uint16_t outputs{ 0 };
202
203 for (const auto &loadEntry : physicalLoadPin)
204 {
205 if (Load::isLocal(loadEntry))
206 bit_set(outputs, Load::pinOf(loadEntry));
207 }
208
209 if constexpr (RELAY_DIVERSION)
210 {
211 for (uint8_t idx = 0; idx < relays.size(); ++idx)
212 bit_set(outputs, relays.get_relay(idx).get_pin());
213 }
214
215 if constexpr (OVERRIDE_PIN_PRESENT)
216 {
217 for (uint8_t i = 0; i < overridePins.size(); ++i)
218 {
219 if (overridePins.getLocalBitmask(i) & ~outputs)
220 return false;
221 }
222 }
223
224 return true;
225}

227constexpr uint16_t check_relay_pins()
228{
229 bool pins_ok{ true };
230
231 for (uint8_t idx = 0; idx < relays.size(); ++idx)
232 {
233 const auto relayPin = relays.get_relay(idx).get_pin();
234
235 if constexpr (RELAY_DIVERSION)
236 {
237 pins_ok &= (relayPin != unused_pin);
238 }
239 else
240 {
241 pins_ok &= (relayPin == unused_pin);
242 }
243 }
244
245 return pins_ok;
