110constexpr PinCheck check_pins()
111{
112 uint16_t used_pins{ 0 };
113
114 if constexpr (TEMP_SENSOR_PRESENT)
115 {
116 if (temperatureSensing.get_pin() != unused_pin)
117 bit_set(used_pins, temperatureSensing.get_pin());
118 }
119
120 if (diversionPin != unused_pin)
121 {
122 if (bit_read(used_pins, diversionPin))
123 return { false, 0 };
124
125 bit_set(used_pins, diversionPin);
126 }
127
128 if (routerOffPin != unused_pin)
129 {
130 if (bit_read(used_pins, routerOffPin))
131 return { false, 0 };
132
133 bit_set(used_pins, routerOffPin);
134 }
135
136 if (rotationPin != unused_pin)
137 {
138 if (bit_read(used_pins, rotationPin))
139 return { false, 0 };
140
141 bit_set(used_pins, rotationPin);
142 }
143
144 if (dualTariffPin != unused_pin)
145 {
146 if (bit_read(used_pins, dualTariffPin))
147 return { false, 0 };
148
149 bit_set(used_pins, dualTariffPin);
150 }
151
152 // Check override pins from the flexible override system
153 if constexpr (OVERRIDE_PIN_PRESENT)
154 {
155 for (uint8_t i = 0; i < overridePins.size(); ++i)
156 {
157 const uint8_t pin = overridePins.getPin(i);
158 if (pin != unused_pin)
159 {
160 if (bit_read(used_pins, pin))
161 return { false, 0 };
162
163 bit_set(used_pins, pin);
164 }
165 }
166 }
167
168 if (watchDogPin != unused_pin)
169 {
170 if (bit_read(used_pins, watchDogPin))
171 return { false, 0 };
172
173 bit_set(used_pins, watchDogPin);
174 }
175
176 // physicalLoadPin: the local TRIACs, plus the optional status LED of each remote load
177 for (const auto &loadEntry : physicalLoadPin)
178 {
179 const uint8_t pin{ Load::pinOf(loadEntry) };
180
181 // A remote load may legitimately have no status LED.
182 // Note that a local entry left at 'unused_pin' (0xFF) decodes as unit 3 / pin 63,
183 // which Load::isValidMap() rejects - the pin field alone cannot carry that sentinel.
184 if (!Load::isLocal(loadEntry) && (pin == 0))
185 continue;
186
187 // a wrong remote unit: Load::hasValidRemoteUnits() reports it
188 if (loadEntry == invalidRemoteUnitEntry)
189 continue;
190
191 // out of range for the bitmask: Load::isValidMap() reports it
192 if (pin > 15)
193 continue;
194
195 if (bit_read(used_pins, pin))
196 return { false, 0 };
197
198 bit_set(used_pins, pin);
199 }
200
201 if constexpr (RELAY_DIVERSION)
202 {
203 for (uint8_t idx = 0; idx < relayDiverter.size(); ++idx)
204 {
205 const auto relayPin = relayDiverter.get_relay(idx).get_pin();
206
207 if (relayPin != unused_pin)
208 {
209 if (bit_read(used_pins, relayPin))
210 return { false, 0 };
211
212 bit_set(used_pins, relayPin);
213 }
214 }
215 }
216
217 return { true, used_pins };
218}

220constexpr bool check_load_priorities()
221{
222 uint8_t _sum{ 0 };
223 uint16_t _enum_val{ 0 };
224
225 for (const auto &loadPrio : loadPrioritiesAtStartup)
226 {
227 // we check if a prio is used twice
228 if (bit_read(_enum_val, loadPrio))
229 return 0;
230
231 bit_set(_enum_val, loadPrio);
232
233 _sum += loadPrio;
234 }
235
236 // check if we have all prio between 0 and (NO_OF_DUMPLOADS - 1)
237 return _sum == ((NO_OF_DUMPLOADS * (NO_OF_DUMPLOADS - 1)) >> 1);
238}

246template< typename List >
247constexpr bool targetsAreOutputs(const List &list)
248{
249 uint16_t outputs{ 0 };
250
251 for (const auto &loadEntry : physicalLoadPin)
252 {
253 if (Load::isLocal(loadEntry))
254 bit_set(outputs, Load::pinOf(loadEntry));
255 }
256
257 if constexpr (RELAY_DIVERSION)
258 {
259 for (uint8_t idx = 0; idx < relayDiverter.size(); ++idx)
260 bit_set(outputs, relayDiverter.get_relay(idx).get_pin());
261 }
262
263 for (uint8_t i = 0; i != list.size(); ++i)
264 {
265 if (list.getLocalBitmask(i) & ~outputs)
266 return false;
267 }
268
269 return true;
270}

283{
284 // a local load on a pin from 16 on lands in the remote range of LOAD(): isValidMap() reports it
285 if (!Load::isValidMap(physicalLoadPin))
286 {
287 return true;
288 }
289
290 const uint16_t existing{ static_cast< uint16_t >((1U << NO_OF_REMOTE_LOADS) - 1) };
291
292 for (uint8_t i = 0; i != list.size(); ++i)
293 {
294 if (list.getRemoteBitmask(i) & ~existing)
295 return false;
296 }
297
298 return true;
