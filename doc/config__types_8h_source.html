156template< uint8_t MaxPins >
157struct PinList
158{
159 uint8_t pins[MaxPins];
160 uint8_t count;
161
165 constexpr PinList()
166 : pins{}, count(0) {}
167
172 constexpr PinList(uint16_t bitmask)
173 : pins{}, count(0)
174 {
175 for (uint8_t pin = 0; pin < 16 && count < MaxPins; ++pin)
176 {
177 if (bitmask & (1U << pin))
178 {
179 pins[count++] = pin; // Store the physical pin number
180 }
181 }
182 }
183
190 constexpr PinList(uint32_t bitmask)
191 : pins{}, count(0)
192 {
193 // Extract local pins from lower 16 bits
194 for (uint8_t pin = 0; pin < 16 && count < MaxPins; ++pin)
195 {
196 if (bitmask & (1UL << pin))
197 {
198 pins[count++] = pin; // Store the physical pin number
199 }
200 }
201 // Extract remote loads from upper 16 bits
202 for (uint8_t i = 0; i < 8 && count < MaxPins; ++i)
203 {
204 if (bitmask & (1UL << (16 + i)))
205 {
206 pins[count++] = REMOTE_PIN_BASE + i; // Store virtual pin for remote load
207 }
208 }
209 }
210
215 template< typename... Args >
216 constexpr PinList(Args... args)
217 : pins{ static_cast< uint8_t >(args)... }, count(sizeof...(args)) {}
218
223 constexpr uint16_t toLocalBitmask() const
224 {
225 uint16_t result{ 0 };
226 for (uint8_t i = 0; i < count; ++i)
227 {
228 if (pins[i] < REMOTE_PIN_BASE)
229 {
230 result |= (1U << pins[i]);
231 }
232 }
233 return result;
234 }
235
241 constexpr uint8_t toRemoteBitmask() const
242 {
243 uint8_t result{ 0 };
244 for (uint8_t i = 0; i < count; ++i)
245 {
246 if (pins[i] >= REMOTE_PIN_BASE)
247 {
248 const uint8_t remoteIndex = pins[i] - REMOTE_PIN_BASE;
249 if (remoteIndex < REMOTE_MASK_BITS)
250 {
251 result |= (1U << remoteIndex);
252 }
253 }
254 }
255 return result;
256 }
257};
