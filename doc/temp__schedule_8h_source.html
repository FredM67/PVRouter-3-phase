163 ++sinceConvert; // the conversion runs in the sensors, late cycles or not
164 }
165 if (sinceConvert < CONVERSION_CYCLES || !onTime)
166 {
167 return Action{ Step::NONE, 0 };

169 Action action{ Step::ADDRESS, sensor };

172 action.step = Step::READ;
173 }
174 else if (2 == sub)
175 {
176 action.step = Step::CLOSE;
177 }
178 if (++sub == STEPS_PER_SENSOR)
179 {
180 sub = 0;
181 ++sensor;
182 }
183 return action;
184 }
