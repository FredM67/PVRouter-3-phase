58template< uint8_t N >
59static double worstWattsError(const float (&cal)[N])
60{
61 const auto fixed{ Datalog::powerCalibration(cal) };
62 double worst{ 0.0 };
63 for (int32_t average = -FULL_SCALE; average <= FULL_SCALE; average += 97)
64 {
65 const double expected{ referenceWatts(average, cal[0]) };
66 if (fabs(expected) > INT16_MAX)
67 {
68 continue; // saturated, tested separately
69 }
70 const double error{ fabs(Datalog::watts(-average, fixed.value[0], fixed.shift) - expected) - fabs(expected) * ldexp(1.0, -16) };
71 if (error > worst)
72 {
73 worst = error;
74 }
75 }
76 return worst;
77}

87template< uint8_t N >
88static double worstVrmsError(uint16_t factor, const float (&cal)[N], uint32_t maxMeanSquare)
89{
90 const auto fixed{ Datalog::voltageCalibration(factor, cal) };
91 double worst{ 0.0 };
92 for (uint8_t phase = 0; phase < N; ++phase)
93 {
94 for (uint32_t meanSquare = 0; meanSquare <= maxMeanSquare; meanSquare += 13)
95 {
96 const double expected{ referenceVrms(meanSquare, factor, cal[phase]) };
97 const double error{ fabs(Datalog::vrms(meanSquare, fixed.value[phase], fixed.shift) - expected) - expected * ldexp(1.0, -14) };
98 if (error > worst)
99 {
100 worst = error;
101 }
102 }
103 }
104 return worst;
105}

115void test_isqrt_at_every_boundary(void)
116{
117 for (uint32_t root = 0; root <= UINT16_MAX; ++root)
118 {
119 const uint32_t square{ root * root };
120 if (root > 1) // r^2 - 1 >= (r - 0.5)^2 from r = 2 on (sqrt(0) = 0)
121 {
122 TEST_ASSERT_EQUAL_UINT16(root, Datalog::isqrt(square - 1));
123 }
124 TEST_ASSERT_EQUAL_UINT16(root, Datalog::isqrt(square));
125 TEST_ASSERT_EQUAL_UINT16(root, Datalog::isqrt(square + root)); // up to UINT32_MAX
126 if (root < UINT16_MAX) // 65535^2 + 65536 = 2^32 wraps
127 {
128 TEST_ASSERT_EQUAL_UINT16(root + 1, Datalog::isqrt(square + root + 1));
129 }
130 }
131}

134void test_isqrt_matches_the_reference(void)
135{
136 constexpr uint32_t STRIDE{ 2147 }; // odd, and not a multiple of small primes: hits all residues
137 uint32_t value{ 0 };
138 for (uint32_t i = 0; i < (UINT32_MAX / STRIDE); ++i, value += STRIDE)
139 {
140 const auto expected{ static_cast< uint32_t >(lround(sqrt(static_cast< double >(value)))) };
141 if (Datalog::isqrt(value) != (expected > UINT16_MAX ? UINT16_MAX : expected))
142 {
143 TEST_FAIL_MESSAGE("isqrt() differs from round(sqrt())");
144 }
145 }
