66{

136template< uint8_t NL, uint8_t NR, uint8_t D >
137class TargetMap
138{
139public:
145 constexpr TargetMap(const uint8_t (&loadMap)[NL], const RelayEngine< NR, D >& relays, const bool relayDiversion)
146 : loadMap_(loadMap), relays_(relays), relayDiversion_(relayDiversion)
147 {
148 }
149
151 constexpr uint8_t pinOf(const uint8_t code) const
152 {
153 if (code >= TargetCode::LOAD && code < TargetCode::LOAD + TargetCode::MAX_LOADS)
154 {
155 const uint8_t loadNum = code - TargetCode::LOAD;
156 return loadNum < NL ? overridePinOf(loadMap_, loadNum) : 0;
157 }
158 if (code >= TargetCode::LOCAL_LOAD && code < TargetCode::LOCAL_LOAD + TargetCode::MAX_LOCAL_LOADS)
159 {
160 const uint8_t loadNum = code - TargetCode::LOCAL_LOAD;
161 return loadNum < NL ? Load::pinOf(loadMap_[loadNum]) : 0;
162 }
163 if (code >= TargetCode::RELAY && code < TargetCode::RELAY + TargetCode::MAX_RELAYS)
164 {
165 const uint8_t relayNum = code - TargetCode::RELAY;
166 return relayDiversion_ && relayNum < NR ? relays_.get_relay(relayNum).pin : 0;
167 }
168 return code;
169 }
170
172 static constexpr bool isGroup(const uint8_t code)
173 {
174 return code >= TargetCode::ALL_LOCAL_LOADS && code <= TargetCode::ALL_LOADS_AND_RELAYS;
175 }
176
178 constexpr uint32_t groupMask(const uint8_t code) const
179 {
180 switch (code)
181 {
182 case TargetCode::ALL_LOCAL_LOADS:
183 return localLoadsMask(loadMap_);
184 case TargetCode::ALL_REMOTE_LOADS:
185 return remoteLoadsMask(loadMap_);
186 case TargetCode::ALL_LOADS:
187 return localLoadsMask(loadMap_) | remoteLoadsMask(loadMap_);
188 case TargetCode::ALL_RELAYS:
189 return relaysMask();
190 case TargetCode::ALL_LOADS_AND_RELAYS:
191 return localLoadsMask(loadMap_) | remoteLoadsMask(loadMap_) | relaysMask();
192 default:
193 return 0;
194 }
195 }

198 template< uint8_t MaxPins >
199 constexpr uint16_t localBitmask(const PinList< MaxPins >& targets) const
200 {
201 uint16_t result{ 0 };
202 for (uint8_t i = 0; i < targets.count; ++i)
203 {
204 const uint8_t code{ targets.pins[i] };
205 result |= isGroup(code) ? static_cast< uint16_t >(groupMask(code)) : PinList< 1 >{ pinOf(code) }.toLocalBitmask();
206 }
207 return result;
208 }
209
211 template< uint8_t MaxPins >
212 constexpr uint8_t remoteBitmask(const PinList< MaxPins >& targets) const
213 {
214 uint8_t result{ 0 };
215 for (uint8_t i = 0; i < targets.count; ++i)
216 {
217 const uint8_t code{ targets.pins[i] };
218 result |= isGroup(code) ? static_cast< uint8_t >(groupMask(code) >> 16) : PinList< 1 >{ pinOf(code) }.toRemoteBitmask();
219 }
220 return result;
221 }
222
223private:
225 constexpr uint32_t relaysMask() const
226 {
227 uint32_t mask{ 0 };
228 if (relayDiversion_)
229 {
230 for (uint8_t i = 0; i < NR; ++i)
231 {
232 bit_set(mask, relays_.get_relay(i).pin);
233 }
234 }
235 return mask;
236 }
237
238 const uint8_t (&loadMap_)[NL];
239 const RelayEngine< NR, D >& relays_;
240 const bool relayDiversion_;
241};
