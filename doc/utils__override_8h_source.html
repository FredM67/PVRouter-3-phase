136template< uint8_t MaxPins >
137struct PinList
138{
139 uint8_t pins[MaxPins];
140 uint8_t count;
141
145 constexpr PinList()
146 : pins{}, count(0) {}
147
152 constexpr PinList(uint16_t bitmask)
153 : pins{}, count(0)
154 {
155 for (uint8_t pin = 0; pin < 16 && count < MaxPins; ++pin)
156 {
157 if (bitmask & (1U << pin))
158 {
159 pins[count++] = pin; // Store the physical pin number
160 }
161 }
162 }
163
170 constexpr PinList(uint32_t bitmask)
171 : pins{}, count(0)
172 {
173 // Extract local pins from lower 16 bits
174 for (uint8_t pin = 0; pin < 16 && count < MaxPins; ++pin)
175 {
176 if (bitmask & (1UL << pin))
177 {
178 pins[count++] = pin; // Store the physical pin number
179 }
180 }
181 // Extract remote loads from upper 16 bits
182 for (uint8_t i = 0; i < 8 && count < MaxPins; ++i)
183 {
184 if (bitmask & (1UL << (16 + i)))
185 {
186 pins[count++] = REMOTE_PIN_BASE + i; // Store virtual pin for remote load
187 }
188 }
189 }

195 template< typename... Args >
196 constexpr PinList(Args... args)
197 : pins{ static_cast< uint8_t >(args)... }, count(sizeof...(args)) {}

203 constexpr uint16_t toLocalBitmask() const
204 {
205 uint16_t result{ 0 };
206 for (uint8_t i = 0; i < count; ++i)
207 {
208 if (pins[i] < REMOTE_PIN_BASE)
209 {
210 result |= (1U << pins[i]);
211 }
212 }
213 return result;
214 }
215
221 constexpr uint8_t toRemoteBitmask() const
222 {
223 uint8_t result{ 0 };
224 for (uint8_t i = 0; i < count; ++i)
225 {
226 if (pins[i] >= REMOTE_PIN_BASE)
227 {
228 const uint8_t remoteIndex = pins[i] - REMOTE_PIN_BASE;
229 if (remoteIndex < 8)
230 {
231 result |= (1U << remoteIndex);
232 }
233 }
234 }
235 return result;
236 }
