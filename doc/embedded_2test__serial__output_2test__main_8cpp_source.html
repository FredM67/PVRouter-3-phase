112/* the JSON format used to be written by ArduinoJson: the same numbers, in the shortest

114void test_printHundredths_matches_arduinojson(void)
115{
116 uint16_t mismatches{ 0 };
117 uint16_t noise{ 0 };
118 char message[80]{};
119
120 for (int32_t value = -5500; value <= 12500; ++value)
121 {
122 char json[24];
123 StaticJsonDocument< 16 > doc;
124 doc.set(static_cast< float >(value) * 0.01F);
125 serializeJson(doc, json, sizeof json);
126
127 capture.clear();
128 SerialOutput::printHundredths(capture, static_cast< int16_t >(value));
129
130 const char* point{ strchr(capture.text, '.') };
131 const size_t decimals{ point ? strlen(point + 1) : 0 };
132 const bool shortest{ decimals <= 2 && (!point || capture.text[capture.length - 1] != '0') };
133 const bool sameValue{ fabs(atof(json) - atof(capture.text)) <= 1e-5 * (1 + fabs(atof(json))) };
134
135 noise += strcmp(json, capture.text) != 0;
136 if (!(shortest && sameValue) && !mismatches++)
137 {
138 snprintf(message, sizeof message, "%ld: ArduinoJson '%s', printHundredths '%s'", static_cast< long >(value), json,
139 capture.text);
140 }
141 }
142 TEST_ASSERT_EQUAL_MESSAGE(0, mismatches, message);
143 TEST_ASSERT_LESS_THAN(1000, noise); // otherwise the two formats really differ
144}
