59constexpr uint8_t crc8(const uint8_t* data, uint8_t len)
60{
61 uint8_t crc{ 0 };
62 while (len--)
63 {
64 uint8_t inbyte{ *data++ };
65 for (uint8_t i = 8; i != 0; --i)
66 {
67 const bool mix{ ((crc ^ inbyte) & 0x01) != 0 };
68 crc >>= 1;
69 if (mix)
70 {
71 crc ^= 0x8C;
72 }
73 inbyte >>= 1;
74 }
75 }
76 return crc;
77}

101constexpr int16_t decode(const ScratchPad& buf)
102{
103 if (crc8(buf, 8) != buf[8])
104 {
105 return DEVICE_DISCONNECTED_RAW;
106 }
107
108 const int16_t result{ toHundredths(buf[0], buf[1]) };
109 if (result < TEMP_RANGE_LOW || result > TEMP_RANGE_HIGH)
110 {
111 return OUTOFRANGE_TEMPERATURE;
112 }
113 return result;
114}

127constexpr int16_t skipPowerOnValue(int16_t reading, int16_t previous)
128{
