45inline int16_t watts(const int32_t averagePower, const uint16_t calFixed, const uint8_t shift)
46{
47 const bool negative{ averagePower < 0 };
48 const uint32_t magnitude{ static_cast< uint32_t >(negative ? -averagePower : averagePower) };
49
50 uint32_t product; // (magnitude x calFixed) >> 8, at most 2^26
51 multU24x16_to32_hi8(product, magnitude, calFixed);
52
53 const uint8_t remaining{ static_cast< uint8_t >(shift - 8) };
54 uint32_t rounded{ (product + (static_cast< uint32_t >(1) << (remaining - 1))) >> remaining };
55 if (rounded > INT16_MAX)
56 {
57 rounded = INT16_MAX;
58 }
59
60 const auto result{ static_cast< int16_t >(rounded) };
61 return negative ? static_cast< int16_t >(-result) : result;
62}

95template< uint8_t SHIFT, typename Count >
96inline int32_t averagePower(const int32_t sum, const Count sampleSets)
97{
98 const auto n{ static_cast< int32_t >(sampleSets) }; // a signed division: the sum is signed
99 if constexpr (SHIFT == 0)
100 {
101 return sum / n;
102 }
103 else
104 {
105 const int32_t quotient{ sum / n };
106 const int32_t remainder{ sum % n }; // |remainder| < n: x 2^SHIFT fits
107 return quotient * (static_cast< int32_t >(1) << SHIFT) + remainder * (static_cast< int32_t >(1) << SHIFT) / n;
108 }
109}

212template< bool Normalise >
213inline uint16_t squareRoot(uint32_t value, uint8_t& fractionBits)
214{
215 uint8_t shift{ 0 };
216 uint24_t remainder;
217 uint24_t twiceRoot;
218
219 if (Normalise && !value)
220 {
221 fractionBits = 0;
222 return 0;
223 }
224
225#ifdef __AVR__
226 uint8_t bytes;
227 uint8_t bits;
228 if constexpr (Normalise)
229 {
230 asm(DATALOG_SQRT_NORMALISE DATALOG_SQRT_DIGITS:DATALOG_SQRT_OPERANDS : : "cc");
231 }
232 else
233 {
234 asm(DATALOG_SQRT_DIGITS:DATALOG_SQRT_OPERANDS : : "cc");
235 }
236#else
237 if constexpr (Normalise)
238 {
239 while (value < (1UL << 24)) // whole bytes, 4 bits of the root each
240 {
241 value <<= 8;
242 shift += 4;
243 }
244 while (value < (1UL << 30)) // then pairs of bits, 1 bit of the root each
245 {
246 value <<= 2;
247 ++shift;
248 }
249 }
250 remainder = 0;
251 twiceRoot = 0;
252 for (uint8_t step = 16; step; --step)
253 {
254 remainder = (remainder << 2) | (value >> 30); // bring down the next two bits
255 value <<= 2;
256 twiceRoot <<= 1; // root <<= 1
257 if (remainder > twiceRoot) // remainder >= 2 x root + 1
258 {
259 remainder -= twiceRoot + 1;
260 twiceRoot |= 2; // root |= 1
261 }
262 }
263#endif
264
265 fractionBits = shift;
266 auto root{ static_cast< uint16_t >(twiceRoot >> 1) };
267 // remainder = value - root^2: round up when value >= (root + 0.5)^2 = root^2 + root + 0.25
268 if ((remainder > root) && (root < UINT16_MAX))
269 {
270 ++root;
271 }
272 return root;
273}

358inline uint16_t vrms(const uint32_t meanSquare, const uint16_t calFixed, const uint8_t shift)
359{
360 uint8_t fractionBits;
361 const uint16_t root{ sqrtScaled(meanSquare, fractionBits) };
362
363 uint32_t product;
364 multU16x16_to32(product, root, calFixed);
365
366 // rounded shift by (fractionBits + shift), without overflowing the 32 bits of product:
367 // whole bytes first (byte moves on AVR), then the remaining bits one by one
368 uint8_t total{ static_cast< uint8_t >(fractionBits + shift) };
369 if (total)
370 {
371 --total; // the last bit is shifted out after the rounding
372 if (total >= 16)
373 {
374 product >>= 16;
375 total -= 16;
376 }
377 if (total >= 8)
378 {
379 product >>= 8;
380 total -= 8;
381 }
382 product = ((product >> total) + 1) >> 1;
383 }
384 return product > UINT16_MAX ? UINT16_MAX : static_cast< uint16_t >(product);
385}
