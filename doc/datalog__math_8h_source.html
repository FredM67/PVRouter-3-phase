44inline int16_t watts(const int32_t averagePower, const uint16_t calFixed, const uint8_t shift)
45{
46 const bool negative{ averagePower < 0 };
47 const uint32_t magnitude{ static_cast< uint32_t >(negative ? -averagePower : averagePower) };
48
49 uint32_t product; // (magnitude x calFixed) >> 8, at most 2^26
50 multU24x16_to32_hi8(product, magnitude, calFixed);
51
52 const uint8_t remaining{ static_cast< uint8_t >(shift - 8) };
53 uint32_t rounded{ (product + (static_cast< uint32_t >(1) << (remaining - 1))) >> remaining };
54 if (rounded > INT16_MAX)
55 {
56 rounded = INT16_MAX;
57 }
58
59 const auto result{ static_cast< int16_t >(rounded) };
60 return negative ? static_cast< int16_t >(-result) : result;
61}

164template< bool Normalise >
165inline uint16_t squareRoot(uint32_t value, uint8_t& fractionBits)
166{
167 uint8_t shift{ 0 };
168 uint24_t remainder;
169 uint24_t twiceRoot;
170
171 if (Normalise && !value)
172 {
173 fractionBits = 0;
174 return 0;
175 }
176
177#ifdef __AVR__
178 uint8_t bytes;
179 uint8_t bits;
180 if constexpr (Normalise)
181 {
182 asm(DATALOG_SQRT_NORMALISE DATALOG_SQRT_DIGITS:DATALOG_SQRT_OPERANDS : : "cc");
183 }
184 else
185 {
186 asm(DATALOG_SQRT_DIGITS:DATALOG_SQRT_OPERANDS : : "cc");
187 }
188#else
189 if constexpr (Normalise)
190 {
191 while (value < (1UL << 24)) // whole bytes, 4 bits of the root each
192 {
193 value <<= 8;
194 shift += 4;
195 }
196 while (value < (1UL << 30)) // then pairs of bits, 1 bit of the root each
197 {
198 value <<= 2;
199 ++shift;
200 }
201 }
202 remainder = 0;
203 twiceRoot = 0;
204 for (uint8_t step = 16; step; --step)
205 {
206 remainder = (remainder << 2) | (value >> 30); // bring down the next two bits
207 value <<= 2;
208 twiceRoot <<= 1; // root <<= 1
209 if (remainder > twiceRoot) // remainder >= 2 x root + 1
210 {
211 remainder -= twiceRoot + 1;
212 twiceRoot |= 2; // root |= 1
213 }
214 }
215#endif
216
217 fractionBits = shift;
218 auto root{ static_cast< uint16_t >(twiceRoot >> 1) };
219 // remainder = value - root^2: round up when value >= (root + 0.5)^2 = root^2 + root + 0.25
220 if ((remainder > root) && (root < UINT16_MAX))
221 {
222 ++root;
223 }
224 return root;
225}

310inline uint16_t vrms(const uint32_t meanSquare, const uint16_t calFixed, const uint8_t shift)
311{
312 uint8_t fractionBits;
313 const uint16_t root{ sqrtScaled(meanSquare, fractionBits) };
314
315 uint32_t product;
316 multU16x16_to32(product, root, calFixed);
317
318 // rounded shift by (fractionBits + shift), without overflowing the 32 bits of product:
319 // whole bytes first (byte moves on AVR), then the remaining bits one by one
320 uint8_t total{ static_cast< uint8_t >(fractionBits + shift) };
321 if (total)
322 {
323 --total; // the last bit is shifted out after the rounding
324 if (total >= 16)
325 {
326 product >>= 16;
327 total -= 16;
328 }
329 if (total >= 8)
330 {
331 product >>= 8;
332 total -= 8;
333 }
334 product = ((product >> total) + 1) >> 1;
335 }
336 return product > UINT16_MAX ? UINT16_MAX : static_cast< uint16_t >(product);
337}
