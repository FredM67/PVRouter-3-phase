109class Reference
110{
111private:
112 static const char STX{ 0x02 };
113 static const char ETX{ 0x03 };
114 static const char LF{ 0x0A };
115 static const char CR{ 0x0D };
116 static const char TAB{ 0x09 };
117
118 char buffer[512]{};
119 size_t bufferPos{ 0 };
120
121 [[nodiscard]] uint8_t calculateChecksum(size_t startPos, size_t endPos) const
122 {
123 uint8_t sum{ 0 };
124 for (const char* ptr = buffer + startPos; ptr != buffer + endPos; ++ptr)
125 {
126 sum += *ptr;
127 }
128
129 return (sum & 0x3F) + 0x20;
130 }
131
132 void writeTag(const char* tag, uint8_t index)
133 {
134 auto* ptr{ tag };
135 while (*ptr) buffer[bufferPos++] = *ptr++;
136
137 if (index != 0)
138 {
139 buffer[bufferPos++] = static_cast< char >('0' + index);
140 }
141
142 buffer[bufferPos++] = TAB;
143 }
144
145 void writeLine(const char* tag, uint8_t index, uint16_t magnitude, bool negative)
146 {
147 buffer[bufferPos++] = LF;
148
149 const auto startPos{ bufferPos };
150
151 writeTag(tag, index);
152 if (negative)
153 {
154 buffer[bufferPos++] = '-';
155 }
156 char* str{ buffer + bufferPos };
157 snprintf(str, 12, "%u", static_cast< unsigned >(magnitude)); // utoa()
158 bufferPos += strlen(str);
159 buffer[bufferPos++] = TAB;
160
161 const auto crc{ calculateChecksum(startPos, bufferPos) };
162 buffer[bufferPos++] = crc;
163
164 buffer[bufferPos++] = CR;
165 }
166
167public:
168 void startFrame()
169 {
170 bufferPos = 0;
171 buffer[bufferPos++] = STX;
172 }
173
174 void send(const char* tag, int16_t value, uint8_t index = 0)
175 {
176 const auto magnitude{ value < 0 ? static_cast< uint16_t >(0U - static_cast< uint16_t >(value)) : static_cast< uint16_t >(value) };
177 writeLine(tag, index, magnitude, value < 0);
178 }
179
180 void sendUnsigned(const char* tag, uint16_t value, uint8_t index = 0)
181 {
182 writeLine(tag, index, value, false);
183 }
184
185 void endFrame()
186 {
187 buffer[bufferPos++] = ETX;
188 }
189
190 [[nodiscard]] const char* data() const
191 {
192 return buffer;
193 }
194 [[nodiscard]] size_t size() const
195 {
196 return bufferPos;
197 }
198};

201template< typename Source >
202static void referenceFrame(Reference& teleInfo)
203{
204 uint8_t idx{ 0 };
205
206 teleInfo.startFrame();
207
208 teleInfo.send("P", Source::power());
209
210 if constexpr (Source::PHASE_COUNT > 1)
211 {
212 idx = Source::PHASE_COUNT;
213 do
214 {
215 --idx;
216 teleInfo.send("P", Source::phasePower(idx), idx + 1);
217 teleInfo.sendUnsigned("V", Source::phaseVoltage(idx), idx + 1); // unsigned, was send() (negative above 327.67 V)
218 } while (idx);
219 }
220
221 if constexpr (Source::RELAY_DIVERSION)
222 {
223 teleInfo.send("R", Source::relayAverage());
224
225 idx = Source::RELAY_COUNT;
226 do
227 {
228 --idx;
229 teleInfo.send("R", Source::relayOn(idx), idx + 1);
230 } while (idx);
231 }
232
233 idx = Source::LOAD_COUNT;
234 do
235 {
236 --idx;
237 teleInfo.send("D", Source::loadOnPercent(idx), idx + 1);
238 } while (idx);
239
240 for (uint8_t idx = 0; idx < Source::TEMP_COUNT; ++idx)
241 {
242 if (!Source::temperatureValid(idx))
243 {
244 continue;
245 }
246 teleInfo.send("T", Source::temperature(idx), idx + 1);
247 }
248
249 teleInfo.sendUnsigned("N", Source::absenceCount()); // unsigned, was send() (negative beyond 32767 s)
250
251 if constexpr (Source::DUAL_TARIFF)
252 {
253 teleInfo.send("TA", static_cast< int16_t >(Source::offPeak() ? 1 : 0));
254 }
255
256 teleInfo.sendUnsigned("S", Source::sampleSets());
257 teleInfo.send("S_MC", Source::lowestSampleSetsPerCycle());
258
259 teleInfo.endFrame();
260}

263struct Streamed

265 char text[512]{};
266 size_t length{ 0 };

271template< typename Source >
272static Streamed streamFrame()
273{
274 Streamed frame;
275 bool more{ true };
276 while (more)
277 {
278 char step[TeleInfo::MAX_STEP_LENGTH];
279 uint8_t length{ 0 };
280 more = TeleInfo::writeFrameStep< Source >(frame.steps++, step, length);
281 TEST_ASSERT_LESS_OR_EQUAL_UINT8(TeleInfo::MAX_STEP_LENGTH, length);
282 memcpy(frame.text + frame.length, step, length);
283 frame.length += length;
284 if (length > frame.longestStep)
285 {
286 frame.longestStep = length;
287 }
288 TEST_ASSERT_LESS_THAN_UINT8(100, frame.steps); // never endless
289 }
290 return frame;
291}

372/* a frame written independently of the code, checksums computed by hand */
373void test_golden_frame(void)
374{
375 values = Values{};
376 values.power = -25;
377 values.phasePower[0] = 399;
378 values.phasePower[1] = 399;
379 values.phasePower[2] = -823;
380 values.phaseVoltage_x100[0] = 23012;
381 values.phaseVoltage_x100[1] = 23100;
382 values.phaseVoltage_x100[2] = 22950;
383 values.loadOnPercent[0] = 40;
384 values.absenceCount = 7;
385 values.sampleSets = 250;
386 values.lowestSampleSetsPerCycle = 44;
387
388 // checksum: the sum of the bytes from the tag to the second TAB, & 0x3F, + 0x20
389 // P\t-25\t 0x0F6 -> 0x36 + 0x20 = 'V'
390 // P3\t-823\t 0x15F -> 0x1F + 0x20 = '?'
391 // V3\t22950\t 0x19D -> 0x1D + 0x20 = '='
392 // P2\t399\t 0x139 -> 0x39 + 0x20 = 'Y'
393 // V2\t23100\t 0x190 -> 0x10 + 0x20 = '0'
394 // P1\t399\t 0x138 -> 0x38 + 0x20 = 'X'
395 // V1\t23012\t 0x191 -> 0x11 + 0x20 = '1'
396 // D1\t40\t 0x0EB -> 0x2B + 0x20 = 'K'
397 // N\t7\t 0x097 -> 0x17 + 0x20 = '7'
398 // S\t250\t 0x0FC -> 0x3C + 0x20 = '\'
399 // S_MC\t44\t 0x1BC -> 0x3C + 0x20 = '\'
400 static const char golden[]{ "\x02"
401 "\nP\t-25\tV\r"
402 "\nP3\t-823\t?\r"
403 "\nV3\t22950\t=\r"
404 "\nP2\t399\tY\r"
405 "\nV2\t23100\t0\r"
406 "\nP1\t399\tX\r"
407 "\nV1\t23012\t1\r"
408 "\nD1\t40\tK\r"
409 "\nN\t7\t7\r"
410 "\nS\t250\t\\\r"
411 "\nS_MC\t44\t\\\r"
412 "\x03" };
413
414 const Streamed streamed{ streamFrame< Minimal >() };
415 TEST_ASSERT_EQUAL_UINT32(sizeof golden - 1, streamed.length);
416 TEST_ASSERT_EQUAL_MEMORY(golden, streamed.text, sizeof golden - 1);
417}

431static uint8_t parseFrame(const Streamed& frame, Line (&lines)[32])
432{
433 TEST_ASSERT_GREATER_THAN_UINT32(2, frame.length);
434 TEST_ASSERT_EQUAL_CHAR(0x02, frame.text[0]);
435 TEST_ASSERT_EQUAL_CHAR(0x03, frame.text[frame.length - 1]);
436
437 uint8_t count{ 0 };
438 size_t pos{ 1 };
439 while (pos < frame.length - 1)
440 {
441 TEST_ASSERT_EQUAL_CHAR('\n', frame.text[pos++]);
442 const size_t start{ pos };
443 const char* const tab1{ static_cast< const char* >(memchr(frame.text + pos, '\t', frame.length - pos)) };
444 TEST_ASSERT_NOT_NULL(tab1);
445 const char* const tab2{ static_cast< const char* >(memchr(tab1 + 1, '\t', frame.length - (tab1 + 1 - frame.text))) };
446 TEST_ASSERT_NOT_NULL(tab2);
447
448 uint8_t sum{ 0 };
449 for (const char* p = frame.text + start; p != tab2 + 1; ++p)
450 {
451 sum += static_cast< uint8_t >(*p);
452 }
453 TEST_ASSERT_EQUAL_CHAR((sum & 0x3F) + 0x20, tab2[1]);
454 TEST_ASSERT_EQUAL_CHAR('\r', tab2[2]);
455
456 Line& line{ lines[count++] };
457 memset(&line, 0, sizeof line);
458 memcpy(line.tag, frame.text + start, tab1 - (frame.text + start));
459 memcpy(line.value, tab1 + 1, tab2 - (tab1 + 1));
460 pos = (tab2 + 3) - frame.text;
461 }
462 return count;
463}

466static uint8_t countTag(const Line (&lines)[32], uint8_t count, const char* tag)
467{
468 uint8_t found{ 0 };
469 for (uint8_t i = 0; i != count; ++i)
470 {
471 found += strcmp(lines[i].tag, tag) == 0 ? 1 : 0;
472 }
473 return found;
474}

477static const char* valueOf(const Line (&lines)[32], uint8_t count, const char* tag)
478{
479 for (uint8_t i = 0; i != count; ++i)
480 {
481 if (strcmp(lines[i].tag, tag) == 0)
482 {
483 return lines[i].value;
484 }
485 }
486 return "";
487}

611/* a single line: LF, tag and index, TAB, value, TAB, checksum, CR */
612void test_line(void)
613{
614 char text[TeleInfo::MAX_STEP_LENGTH];
615
616 // T1\t-5500\t: 0x18E -> 0x0E + 0x20 = '.'
617 const char* end{ TeleInfo::line(text, "T", -5500, 1) };
618 TEST_ASSERT_EQUAL_UINT32(12, end - text);
619 TEST_ASSERT_EQUAL_MEMORY("\nT1\t-5500\t.\r", text, 12);
620
621 // V1\t65535: unsigned, not -1
622 end = TeleInfo::lineUnsigned(text, "V", 65535, 1);
623 TEST_ASSERT_EQUAL_UINT32(TeleInfo::lineSize(2, 5), end - text);
624 TEST_ASSERT_EQUAL_MEMORY("\nV1\t65535\t", text, 10);
625
626 // the longest: a 4-character tag and a signed 5-digit value
627 end = TeleInfo::line(text, "S_MC", -32768);
628 TEST_ASSERT_EQUAL_UINT32(TeleInfo::lineSize(4, 6), end - text);
629 TEST_ASSERT_EQUAL_UINT8(TeleInfo::MAX_STEP_LENGTH - 1, TeleInfo::lineSize(4, 6));
630
631 // 0 and -32768
632 end = TeleInfo::line(text, "P", 0);
633 TEST_ASSERT_EQUAL_MEMORY("\nP\t0\t", text, 5);
634 end = TeleInfo::line(text, "P", -32768);
635 TEST_ASSERT_EQUAL_MEMORY("\nP\t-32768\t", text, 10);
636}
