52inline uint32_t end()
53{
54 const uint8_t sreg{ SREG };
55 cli();
56 // read while counting: once stopped, simavr reads TCNT1 as 0
57 const uint16_t low{ TCNT1 };
58 uint16_t high{ overflows };
59 if ((TIFR1 & _BV(TOV1)) && low < 0x8000U)
60 {
61 ++high; // overflow between the last interrupt and the read, not served yet
62 }
63 TCCR1B = 0;
64 TIMSK1 = 0;
65 TIFR1 = _BV(TOV1);
66 TIMSK0 = savedTimsk0;
67 SREG = sreg;
68 return (static_cast< uint32_t >(high) << 16) | low;
69}
