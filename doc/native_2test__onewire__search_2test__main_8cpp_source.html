24class SimulatedBus
25{
26public:
27 void add(const uint8_t (&rom)[8])
28 {
29 if (count != MAX_DEVICES)

33 }

41 bool reset()
42 {
43 ++resets;
44 for (uint8_t i = 0; i != count; ++i)
45 {
46 inSearch[i] = true;
47 }
48 bit = 0;
49 readComplement = false;
50 return count != 0; // presence pulse
51 }
52
53 void write(uint8_t command)
54 {
55 lastCommand = command;
56 }
57
58 uint8_t read_bit()
59 {
60 uint8_t line{ 1 };
61 for (uint8_t i = 0; i != count; ++i)
62 {
63 if (inSearch[i] && (bitOf(i) == readComplement)) // sends 0: bit 0, or complement of 1
64 {
65 line = 0;
66 }
67 }
68 readComplement = !readComplement;
69 return line;
70 }
71
72 void write_bit(uint8_t value)
73 {
74 for (uint8_t i = 0; i != count; ++i)
75 {
76 if (bitOf(i) != (value != 0))
77 {
78 inSearch[i] = false;
79 }
80 }
81 ++bit;
82 }
83
85 void unplugAll()
86 {
87 plugged = count;
88 count = 0;
89 }

91 uint8_t plugged{ 0 };
92
93 uint8_t lastCommand{ 0 };
94 uint16_t resets{ 0 };
95
96private:
97 bool bitOf(uint8_t device) const
98 {
99 return (roms[device][bit / 8] >> (bit % 8)) & 1U;
100 }

102 uint8_t roms[MAX_DEVICES][8]{};
103 bool inSearch[MAX_DEVICES]{};
104 uint8_t count{ 0 };
105 uint8_t bit{ 0 };
106 bool readComplement{ false };
107};

110uint8_t searchAll(SimulatedBus& bus, uint8_t (&found)[MAX_DEVICES + 1][8])
111{
112 RomSearch< SimulatedBus > search{ bus };
113 uint8_t rom[8]{};
114 uint8_t n{ 0 };
115 while ((n != MAX_DEVICES + 1) && search.next(rom))
116 {
117 memcpy(found[n++], rom, 8);
118 }
119 return n;
120}

122bool contains(const uint8_t (&found)[MAX_DEVICES + 1][8], uint8_t n, const uint8_t* rom)
123{
124 for (uint8_t i = 0; i != n; ++i)
125 {
126 if (0 == memcmp(found[i], rom, 8))
127 {
128 return true;
129 }
130 }
131 return false;
132}
