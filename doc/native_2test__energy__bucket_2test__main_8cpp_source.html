33static double quantisationError(float cal, uint8_t shift)
34{
35 const double exact{ static_cast< double >(cal) * ldexp(1.0, shift) };
36 return fabs(Energy::toFixed(cal, shift) - exact) / exact;

274void test_per_sample_agrees_with_the_division_path(void)
275{
276 // the fallback (out-of-range n) divides first: both paths must agree within the
277 // truncation of that division (< 1 raw unit, i.e. < 1 bucket unit), their two
278 // roundings, and their two calibration quantisations (which may point opposite ways)
279 constexpr auto table{ Energy::toFixedPerSample< N_MIN_50HZ, N_MAX_50HZ >(typicalCal) };
280 constexpr uint8_t shift{ Energy::calibrationShift(typicalCal) };
281 constexpr uint16_t calFixed{ Energy::toFixed(typicalCal[0], shift) };
282
283 for (uint8_t n = N_MIN_50HZ; n <= N_MAX_50HZ; ++n)
284 {
285 for (int32_t sumP = -static_cast< int32_t >(n) * FULL_SCALE; sumP <= static_cast< int32_t >(n) * FULL_SCALE; sumP += 77777)
286 {
287 const int32_t direct{ Energy::contribution(sumP, table.value[0][n - N_MIN_50HZ], table.shift) };
288 const int32_t divided{ Energy::contribution(sumP / n, calFixed, shift) };
289
290 TEST_ASSERT_TRUE_MESSAGE(fabs(static_cast< double >(direct - divided)) <= 2.0 + fabs(static_cast< double >(divided)) * 6e-5, "the per-sample and division paths disagree");
291 }
292 }

295void test_per_sample_fallback_for_an_unexpected_cycle_length(void)
296{
297 // n outside the table (start-up, missing phase): the ISR rescales the sum to NMin
298 // samples, (sumP / n) x NMin, and uses the NMin entry - same shift, no second path
299 constexpr auto table{ Energy::toFixedPerSample< N_MIN_50HZ, N_MAX_50HZ >(typicalCal) };
300 static constexpr uint8_t unexpected[]{ 1, 5, 20, 27, 37, 64, 100, 255 };
301
302 for (const uint8_t n : unexpected)
303 {
304 for (int32_t average = -FULL_SCALE; average <= FULL_SCALE; average += 9973)
305 {
306 // with a remainder, so that the division truncates as it does in the ISR
307 const int32_t sumP{ average * n + ((average < 0) ? -(n - 1) : (n - 1)) };
308 const int32_t rescaled{ (sumP / n) * N_MIN_50HZ };
309 const double expected{ exactReference(sumP, n, typicalCal[1]) };
310 const double actual{ static_cast< double >(Energy::contribution(rescaled, table.value[1][0], table.shift)) };
311
312 // one unit of rounding, < 1 unit of truncation, plus the quantisation
313 TEST_ASSERT_TRUE_MESSAGE(fabs(actual - expected) <= 2.0 + fabs(expected) * 4e-5, "the fallback differs from the exact reference");
314 }
315 }

372/* the exact value, rounded half away from zero, over the whole bucket and beyond */
373template< uint8_t SUPPLY_FREQUENCY >
374void checkHundredthsOfJoule()
375{
376 const int32_t capacity{ Energy::fromWatts(3600L * SUPPLY_FREQUENCY) };
377 const double unitsPerJoule{ static_cast< double >(SUPPLY_FREQUENCY << Energy::FRACTION_BITS) };
378
379 for (int32_t units = -2 * capacity; units <= 2 * capacity; units += 7)
380 {
381 const auto expected{ static_cast< int32_t >(llround(units * 100.0 / unitsPerJoule)) };
382 if (expected != Energy::toHundredthsOfJoule< SUPPLY_FREQUENCY >(units))
383 {
384 TEST_ASSERT_EQUAL_INT32_MESSAGE(expected, Energy::toHundredthsOfJoule< SUPPLY_FREQUENCY >(units), "units");
385 }
386 }
387}
