394 void proceed_relays(const uint16_t overrideBitmask, const bool diversionEnabled = true) const
395 {
396 if (settle_change != 0)
397 {
398 // A relay has been toggle less than a minute ago, wait until changes take effect
399 return;
400 }
401
402 if (!diversionEnabled)
403 {
404 bool anyChanges{ false };
405 uint8_t idx{ N };
406 do
407 {
408 anyChanges |= relay[--idx].proceed_forced_only(overrideBitmask);
409 } while (idx);
410
411 if (anyChanges)
412 settle_change = 60;
413
414 return;
415 }
416
417 const auto currentAvgPower{ ewma_average.getAverageT() };
418
419 const bool importing{ currentAvgPower > 0 };
420
421 // First from the last relay down: OFF on an import (where a forced relay comes in too), and
422 // on a small surplus for the relays in battery mode. Then, on a surplus only, from the first
423 // relay up: ON. One call site, so that the switching code is inlined once.
424 for (uint8_t step = 0; step != 2 * N; ++step)
425 {
426 const bool down{ step < N };
427 if (!down && importing)
428 {
429 return;
430 }
431
432 const uint8_t idx{ static_cast< uint8_t >(down ? N - 1 - step : step - N) };
433 if (relay[idx].proceed_relay(currentAvgPower, overrideBitmask, importing || !down))
434 {
435 settle_change = 60;
436 return;
437 }
438 }
439 }
