48uint16_t simulateADC(float voltage_inst)
49{
50 float adc = (voltage_inst / 400.0f) * 1024.0f + 512.0f;
51 if (adc < 0) adc = 0;
52 if (adc > 1023) adc = 1023;
53 return static_cast< uint16_t >(adc) << 6;

67void test_voltage_accumulation(uint8_t period, uint8_t shift, float vrms, bool expect_overflow, const char* test_name)
68{
69 const float vpeak = vrms * 1.414f;
70 const uint32_t cycles = period * SUPPLY_FREQUENCY;
71 const uint32_t samples = static_cast< uint32_t >(cycles * (20000.0f / SAMPLE_PERIOD_US));
72
73 if constexpr (use_64bit)
74 {
75 // Use 64-bit accumulator to detect overflow
76 uint64_t sum_vsq = 0;
77 uint64_t max_acc = 0;
78 float time_us = 0.0f;
79
80 for (uint32_t s = 0; s < samples; ++s)
81 {
82 float angle = 2.0f * PI * SUPPLY_FREQUENCY * time_us / 1e6f;
83 uint16_t adc = simulateADC(vpeak * sin(angle));
84 int16_t sample = (adc | 32U) - ADC_MIDPOINT_ALIGNED;
85 // Optimization: reduce to 14-bits before multiplication (more efficient on 8-bit MCU)
86 int16_t sample_div4 = sample >> 2; // reduce to 14-bits (now x16, or 2^4)
87 int32_t vsq_signed;
88 multS16x16_to32(vsq_signed, sample_div4, sample_div4); // 32-bits (now x256, or 2^8)
89 uint32_t vsq = static_cast< uint32_t >(vsq_signed); // V² is always positive
90 vsq >>= shift;
91 sum_vsq += vsq;
92 if (sum_vsq > max_acc) max_acc = sum_vsq;
93 time_us += SAMPLE_PERIOD_US;
94 }
95
96 Serial.print(test_name);
97 Serial.print(F(": samples="));
98 Serial.print(samples);
99 Serial.print(F(", max="));
100 printU64(max_acc);
101 Serial.print(F(" vs UINT32_MAX="));
102 Serial.println(UINT32_MAX);
103
104 // Unity's *_UINT64 assertions are compiled out on AVR (they expand to an
105 // unconditional "64-bit Support Disabled" failure), so compare here and
106 // assert on the resulting bool. The uint64_t arithmetic itself is fine.
107 const bool overflows{ max_acc > static_cast< uint64_t >(UINT32_MAX) };
108
109 if (expect_overflow)
110 {
111 // Accumulator exceeds UINT32_MAX, i.e. a uint32_t accumulator would wrap here
112 TEST_ASSERT_TRUE_MESSAGE(overflows, "expected uint32_t overflow, but the sum still fits");
113 }
114 else
115 {
116 TEST_ASSERT_FALSE_MESSAGE(overflows, "unexpected uint32_t overflow");
117 }
118 }
119 else
120 {
121 // Use 32-bit accumulator (normal case)
122 uint32_t sum_vsq = 0;
123 uint32_t max_acc = 0;
124 float time_us = 0.0f;
125
126 for (uint32_t s = 0; s < samples; ++s)
127 {
128 float angle = 2.0f * PI * SUPPLY_FREQUENCY * time_us / 1e6f;
129 uint16_t adc = simulateADC(vpeak * sin(angle));
130 int16_t sample = (adc | 32U) - ADC_MIDPOINT_ALIGNED;
131 // Optimization: reduce to 14-bits before multiplication (more efficient on 8-bit MCU)
132 int16_t sample_div4 = sample >> 2; // reduce to 14-bits (now x16, or 2^4)
133 int32_t vsq_signed;
134 multS16x16_to32(vsq_signed, sample_div4, sample_div4); // 32-bits (now x256, or 2^8)
135 uint32_t vsq = static_cast< uint32_t >(vsq_signed); // V² is always positive
136 vsq >>= shift;
137 sum_vsq += vsq;
138 if (sum_vsq > max_acc) max_acc = sum_vsq;
139 time_us += SAMPLE_PERIOD_US;
140 }
141
142 uint32_t headroom = UINT32_MAX - max_acc;
143
144 Serial.print(test_name);
145 Serial.print(F(": samples="));
146 Serial.print(samples);
147 Serial.print(F(", max="));
148 Serial.print(max_acc);
149 Serial.print(F(", headroom="));
150 Serial.println(headroom);
151
152 // Verify we have significant headroom (at least 10% of UINT32_MAX)
153 // This ensures we're not close to overflow
154 TEST_ASSERT_GREATER_THAN_UINT32(UINT32_MAX / 10, headroom);
155 }
