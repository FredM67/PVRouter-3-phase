53template< uint8_t NL >
54constexpr uint32_t loadsOf(const uint8_t code, const uint8_t (&loadMap)[NL])
55{
56 static_assert(NL <= 32, "the load numbers of the rotation groups fit 32 bits");
57
58 if (code >= TargetCode::LOAD && code < TargetCode::LOAD + TargetCode::MAX_LOADS)
59 {
60 const uint8_t loadNum = code - TargetCode::LOAD;
61 return loadNum < NL ? (1UL << loadNum) : 0;
62 }
63 if (code >= TargetCode::LOCAL_LOAD && code < TargetCode::LOCAL_LOAD + TargetCode::MAX_LOCAL_LOADS)
64 {
65 const uint8_t loadNum = code - TargetCode::LOCAL_LOAD;
66 return loadNum < NL && Load::isLocal(loadMap[loadNum]) ? (1UL << loadNum) : 0;
67 }
68 if (code >= TargetCode::RELAY && code < TargetCode::RELAY + TargetCode::MAX_RELAYS)
69 {
70 return 0;
71 }
72
73 uint32_t loads{ 0 };
74 if (code == TargetCode::ALL_LOCAL_LOADS || code == TargetCode::ALL_REMOTE_LOADS || code == TargetCode::ALL_LOADS)
75 {
76 for (uint8_t i = 0; i != NL; ++i)
77 {
78 const bool local = Load::isLocal(loadMap[i]);
79 if (code == TargetCode::ALL_LOADS || local == (code == TargetCode::ALL_LOCAL_LOADS))
80 {
81 loads |= 1UL << i;
82 }
83 }
84 return loads;
85 }
86 if (code >= TargetCode::ALL_LOCAL_LOADS)
87 {
88 return 0; // ALL_RELAYS(), ALL_LOADS_AND_RELAYS()
89 }
90
91 if (code >= REMOTE_PIN_BASE)
92 {
93 uint8_t ordinal = code - REMOTE_PIN_BASE;
94 for (uint8_t i = 0; i != NL; ++i)
95 {
96 if (!Load::isLocal(loadMap[i]) && ordinal-- == 0)
97 {
98 return 1UL << i;
99 }
100 }
101 return 0;
102 }
103
104 for (uint8_t i = 0; i != NL; ++i)
105 {
106 if (Load::isLocal(loadMap[i]) && Load::pinOf(loadMap[i]) == code)
107 {
108 return 1UL << i;
109 }
110 }
111 return 0;
112}

181template< uint8_t NG, uint8_t MaxPins, uint8_t NL >
182constexpr bool groupsAreConsistent(const RotationGroups< NG, MaxPins >& groups, const uint8_t (&loadMap)[NL])
183{
184 for (uint8_t g = 0; g != groups.size(); ++g)
185 {
186 const uint32_t loads = groupLoads(groups, g, loadMap);
187 bool first{ true };
188 uint8_t unit{ 0 };
189 for (uint8_t i = 0; i != NL; ++i)
190 {
191 if (!(loads & (1UL << i)))
192 {
193 continue;
194 }
195 if (first)
196 {
197 unit = Load::unitOf(loadMap[i]);
198 first = false;
199 }
200 else if (Load::unitOf(loadMap[i]) != unit)
201 {
202 return false;
203 }
204 }
205 }
206 return true;

232 {

261 template< uint8_t N >
262 __attribute__((always_inline)) inline void rotate(uint8_t (&prio)[N]) const
263 {
264 uint8_t start{ 0 };
265 for (uint8_t g = 0; g != NG; ++g)
266 {
267 const uint8_t end{ ends_[g] };
268 if (end - start >= 2)
269 {
270 const uint8_t last{ prio[places_[end - 1]] };
271 for (uint8_t i = end - 1; i != start; --i)
272 {
273 prio[places_[i]] = prio[places_[i - 1]];
274 }
275 prio[places_[start]] = last;
276 }
277 start = end;
278 }
279 }
