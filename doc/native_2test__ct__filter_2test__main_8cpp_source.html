27struct Robin
28{
29 int32_t lpf_long{};
30
31 int32_t apply(const int16_t countsMinusDC, const float lpf_gain, const float alpha)
32 {
33 int32_t sampleIminusDC_grid = ((int32_t)countsMinusDC) << 8;
34 const int32_t last_lpf_long = lpf_long;
35 lpf_long += alpha * (sampleIminusDC_grid - last_lpf_long);
36 sampleIminusDC_grid += (lpf_gain * lpf_long);
37 return sampleIminusDC_grid;
38 }
39};

62static int16_t sineCounts(const uint16_t i, const int16_t dcResidual, const int16_t amplitudeFactor)

72static int16_t inputCounts(const Shape shape, const uint16_t i, uint32_t &seed, const float gain)
73{
74 switch (shape)
75 {
76 case Shape::STEP:
77 return (i < 100) ? 0 : static_cast< int16_t >(30000.0F / (64.0F * (1.0F + gain))); // output within int16
78 case Shape::SINE:
79 return sineCounts(i, 5, 2); // 200 counts peak, 5 counts of DC residual
80 default:
81 seed = seed * 1664525UL + 1013904223UL;
82 return static_cast< int16_t >(static_cast< int16_t >((seed >> 16) % 301U) - 150 + 3); // -147..153
83 }

87static int32_t maxOutputError(const Shape shape, const float gain, const float alpha, const uint16_t samples)
88{
89 const CtFilter::Coefficients coefficients{ CtFilter::coefficientsOf(gain, alpha) };
90 CtFilter::LowPassBoost filter;
91 Robin robin;
92 uint32_t seed{ 12345 };
93 int32_t worst{ 0 };
94
95 for (uint16_t i = 0; i != samples; ++i)
96 {
97 const int16_t counts{ inputCounts(shape, i, seed, gain) };
98 const int32_t expected{ robin.apply(counts, gain, alpha) >> 2 }; // back to x64
99 const int32_t actual{ filter.apply(x64(counts), coefficients) };
100 const int32_t error{ (actual > expected) ? actual - expected : expected - actual };
101 if (error > worst)
102 {
103 worst = error;
104 }
105 }
106 return worst;
107}

129void test_saturates_at_the_int16_limits(void)
130{
131 const CtFilter::Coefficients coefficients{ CtFilter::coefficientsOf(9.0F, 0.0011F) };
132 // a DC residual of 6 counts builds up the low-pass state...
133 CtFilter::LowPassBoost positive;
134 CtFilter::LowPassBoost negative;
135 for (uint16_t i = 0; i != 20000; ++i)
136 {
137 positive.apply(x64(6), coefficients);
138 negative.apply(x64(-6), coefficients);
139 }
140 TEST_ASSERT_TRUE(positive.lowPass() > 0);
141 TEST_ASSERT_TRUE(negative.lowPass() < 0);
142
143 // ... then a current peak at full scale: beyond int16, it saturates with its sign kept
144 TEST_ASSERT_EQUAL_INT16(INT16_MAX, positive.apply(x64(511), coefficients));
145 TEST_ASSERT_EQUAL_INT16(INT16_MIN, negative.apply(x64(-512), coefficients));

161void test_time_constant_within_2_percent_of_robins(void)
162{
163 // After a step, the low-pass reaches 1 - 1/e of it in -1 / ln(1 - alpha) samples
164 static constexpr float alphas[]{ 0.0011F, 0.002F };
165 for (const float alpha : alphas)
166 {
167 const CtFilter::Coefficients coefficients{ CtFilter::coefficientsOf(9.0F, alpha) };
168 const float expected{ static_cast< float >(-1.0 / log(1.0 - static_cast< double >(alpha))) };
169
170 static constexpr int16_t STEP{ 200 }; // counts
171 CtFilter::LowPassBoost filter;
172 uint16_t samples{ 0 };
173 while (lowPassCounts(filter) < static_cast< float >(STEP * (1.0 - exp(-1.0))))
174 {
175 filter.apply(x64(STEP), coefficients);
176 ++samples;
177 TEST_ASSERT_TRUE(samples < 10000);
178 }
179 TEST_ASSERT_FLOAT_WITHIN(expected * 0.02F, expected, static_cast< float >(samples));
180 }
181}
