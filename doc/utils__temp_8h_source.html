241 [[nodiscard]] int16_t readTemperature(const uint8_t idx) const
242 {
243 if constexpr (!TEMP_SENSOR_PRESENT)
244 {
245 // Suppress unused parameter warning when temperature sensing is disabled
246 (void)idx;
247 return 0;
248 }
249 else
250 {
251 static ScratchPad buf;
252
253 if (!oneWire.reset())
254 {
255 return DEVICE_DISCONNECTED_RAW;
256 }
257 oneWire.select(sensorAddrs[idx].addr);
258 oneWire.write(READ_SCRATCHPAD);
259
260 for (auto& buf_elem : buf)
261 {
262 buf_elem = oneWire.read();
263 }
264
265 if (!oneWire.reset())
266 {
267 return DEVICE_DISCONNECTED_RAW;
268 }
269 if (oneWire.crc8(buf, 8) != buf[8])
270 {
271 return DEVICE_DISCONNECTED_RAW;
272 }
273
274 // result is temperature x16, multiply by 6.25 to convert to temperature x100
275 int16_t result{ (buf[1] << 8) | buf[0] };
276 result = (result * 6) + (result >> 2);
277 if (result <= TEMP_RANGE_LOW || result >= TEMP_RANGE_HIGH)
278 {
279 return OUTOFRANGE_TEMPERATURE; // return value ('Out of range')
280 }
281
282 return result;
283 }
284 }
