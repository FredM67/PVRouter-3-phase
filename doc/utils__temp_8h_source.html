141 void printSensorsOnBus() const
142 {
143 if constexpr (TEMP_SENSOR_PRESENT && CALIBRATION_MODE)
144 {
145 oneWire.begin(sensorPin);
146
147 info(F("Temperature sensors found on pin "));
148 info(sensorPin);
149 infoln(F(":"));
150
151 RomSearch< OneWireType > search{ oneWire };
152 DeviceAddress found;
153 uint8_t count{ 0 };
154 // bounded: noise on a long or badly terminated bus could otherwise keep the search going
155 constexpr uint8_t MAX_LISTED{ 16 };
156 static_assert(MAX_LISTED == 16, "update the message below");
157 while (search.next(found.addr))
158 {
159 if (MAX_LISTED == count)
160 {
161 // a 17th device: exactly 16 sensors give no warning
162 infoln(F("\t... more than 16: noise on the bus? Check the wiring"));
163 break;
164 }
165 ++count;
166 info(F("\t{ "));
167 for (uint8_t i = 0; i != 8; ++i)
168 {
169 info(F("0x"));
170 if (found.addr[i] < 0x10)
171 {
172 info('0');
173 }
174 info(found.addr[i], HEX);
175 if (i != 7)
176 {
177 info(F(", "));
178 }
179 }
180 info(F(" }"));
181
182 if (DS18B20::crc8(found.addr, 7) != found.addr[7])
183 {
184 infoln(F(" CRC error: check the wiring"));
185 }
186 else
187 {
188 infoln(isConfigured(found) ? F(" in config.h") : F(" not in config.h"));
189 }
190 }
191 if (0 == count)
192 {
193 infoln(F("\tnone: check the wiring and the pull-up resistor"));
194 }
195 }

220 [[nodiscard]] bool startRead(const uint8_t idx) const
221 {
222 if constexpr (!TEMP_SENSOR_PRESENT)
223 {
224 (void)idx;
225 return false;
226 }
227 else
228 {
229 if (!oneWire.reset())
230 {
231 return false;
232 }
233 oneWire.write(MATCH_ROM);
234 for (uint8_t i = 0; i != ROM_HALF; ++i)
235 {
236 oneWire.write(sensorAddrs[idx].addr[i]);
237 }
238 return true;
239 }

248 void readScratchpad(const uint8_t idx) const
249 {
250 if constexpr (!TEMP_SENSOR_PRESENT)
251 {
252 (void)idx;
253 }
254 else
255 {
256 for (uint8_t i = ROM_HALF; i != sizeof(sensorAddrs[idx].addr); ++i)
257 {
258 oneWire.write(sensorAddrs[idx].addr[i]);
259 }
260 oneWire.write(READ_SCRATCHPAD);
261
262 for (auto& buf_elem : scratchPad)
263 {
264 buf_elem = oneWire.read();
265 }
266 }
