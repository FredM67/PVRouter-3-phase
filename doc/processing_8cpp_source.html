172constexpr uint16_t getOutputPins()
173{
174 uint16_t output_pins{ 0 };
175
176 // Local TRIAC pins, plus the optional status LED of each remote load
177 for (const auto &loadEntry : physicalLoadPin)
178 {
179 const uint8_t pin{ Load::pinOf(loadEntry) };
180
181 if (!Load::isLocal(loadEntry) && (pin == 0))
182 continue; // remote load without a status LED
183
184 if (bit_read(output_pins, pin))
185 return 0;
186
187 bit_set(output_pins, pin);
188 }
189
190 if constexpr (WATCHDOG_PIN_PRESENT)
191 {
192 if (bit_read(output_pins, watchDogPin))
193 return 0;
194
195 bit_set(output_pins, watchDogPin);
196 }
197
198 if constexpr (RELAY_DIVERSION)
199 {
200 for (uint8_t idx = 0; idx < relays.size(); ++idx)
201 {
202 const auto relayPin = relays.get_relay(idx).get_pin();
203
204 if (bit_read(output_pins, relayPin))
205 return 0;
206
207 bit_set(output_pins, relayPin);
208 }
209 }
210
211 return output_pins;

294void initializeProcessing()
295{
296 // Initialize DC offset in Q16.16 format: (0x8000 << 16) for left-aligned ADC
297 // 0x8000 is the mid-point of a left-aligned 10-bit ADC result
298 initializeArray(i_DCoffset_V, i_DCoffset_V_nom); // nominal mid-point value of ADC left-aligned, x64 scale
299
300 constexpr uint32_t filterDC_V_nom = static_cast< uint32_t >(i_DCoffset_V_nom) << 15;
301 initializeArray(l_filterDC_V, filterDC_V_nom); // nominal mid-point value of ADC left-aligned
302
303 setPinsAsOutput(getOutputPins()); // set the output pins as OUTPUT
304 setPinsAsInputPullup(getInputPins()); // set the input pins as INPUT_PULLUP
305
306 uint8_t i{ NO_OF_DUMPLOADS };
307 do
308 {
309 --i;
310 loadPrioritiesAndState[i] = loadPrioritiesAtStartup[i];
311 loadPrioritiesAndState[i] &= loadStateMask;
312 } while (i);
313
314 if constexpr (RF_CHIP_PRESENT)
315 {
316 // Initialize shared RF module
317 if (initialize_rf())
318 {
319 infoln(F("RF module initialized"));
320 }
321 else
322 {
323 infoln(F("RF module initialization FAILED"));
324 }
325 }
326
327 if constexpr (REMOTE_LOADS_PRESENT)
328 {
329 remoteLoads.reset();
330 infoln(F("Remote loads initialized"));
331 }
332
333 // First stop the ADC
334 bit_clear(ADCSRA, ADEN);
335
336 // Activate free-running mode
337 ADCSRB = 0x00;
338
339 // Reference and left-aligned result for the first conversion too, on the first channel (V1)
340 ADMUX = _ADMUX | sensorV[0];
341
342 // Set up the ADC to be free-running
343 bit_set(ADCSRA, ADPS0); // Set the ADC's clock to system clock / 128
344 bit_set(ADCSRA, ADPS1);
345 bit_set(ADCSRA, ADPS2);
346
347 bit_set(ADCSRA, ADATE); // set the Auto Trigger Enable bit in the ADCSRA register. Because
348 // bits ADTS0-2 have not been set (i.e. they are all zero), the
349 // ADC's trigger source is set to "free running mode".
350
351 bit_set(ADCSRA, ADIE); // set the ADC interrupt enable bit. When this bit is written
352 // to one and the I-bit in SREG is set, the
353 // ADC Conversion Complete Interrupt is activated.
354
355 bit_set(ADCSRA, ADEN); // Enable the ADC
356
357 bit_set(ADCSRA, ADSC); // start ADC manually first time
358
359 sei(); // Enable Global Interrupts
360}

377void updatePortsStates()
378{
379 uint16_t pinsON{ 0 };
380 uint16_t pinsOFF{ 0 };
381
382 // One pass over the load map. A local load drives its TRIAC pin; a remote load is
383 // driven over RF and only drives its optional status LED here, if it has one.
384 uint8_t i{ NO_OF_DUMPLOADS };
385 do
386 {
387 --i;
388
389 const uint8_t loadEntry{ physicalLoadPin[i] };
390 // with an all-local map both of these fold away at compile time, leaving the original code
391 const uint8_t pin{ REMOTE_LOADS_PRESENT ? Load::pinOf(loadEntry) : loadEntry };
392 const bool bDrivesAPin{ !REMOTE_LOADS_PRESENT || Load::isLocal(loadEntry) || (pin != 0) };
393
394 if (LoadStates::LOAD_OFF == physicalLoadState[i])
395 {
396 if (bDrivesAPin)
397 {
398 pinsOFF |= bit(pin);
399 }
400 }
401 else
402 {
403 // count ON-cycles on the load's *state*, so remote loads are datalogged too
404 ++countLoadON[i];
405
406 if (bDrivesAPin)
407 {
408 pinsON |= bit(pin);
409 }
410 }
411 } while (i);
412
413 setPinsOFF(pinsOFF);
414 setPinsON(pinsON);
415}

441void updatePhysicalLoadStates()
442{
443 if constexpr (NO_OF_DUMPLOADS == 0)
444 {
445 return;
446 }
447
448 if constexpr (PRIORITY_ROTATION != RotationModes::OFF)
449 {
450 if (Shared::b_reOrderLoads)
451 {
452 uint8_t i{ static_cast< uint8_t >(NO_OF_DUMPLOADS - 1) };
453 const auto temp{ loadPrioritiesAndState[i] };
454 do
455 {
456 loadPrioritiesAndState[i] = loadPrioritiesAndState[i - 1];
457 --i;
458 } while (i);
459 loadPrioritiesAndState[0] = temp;
460
461 Shared::b_reOrderLoads = false;
462 }
463 }
464
465 // Diversion OFF stops the surplus diversion only: a forced load stays ON. With the router
466 // OFF, the main loop clears the overrides and disables diversion: every load is OFF.
467 const bool bDiversionEnabled{ Shared::b_diversionEnabled };
468 uint8_t idx{ NO_OF_DUMPLOADS };
469 do
470 {
471 --idx;
472 const auto iLoad{ loadPrioritiesAndState[idx] & loadStateMask };
473 const uint8_t loadEntry{ physicalLoadPin[iLoad] };
474
475 // Check override based on load type (local vs remote).
476 // With an all-local map the whole test folds down to the local branch.
477 const bool bOverrideActive{
478 (!REMOTE_LOADS_PRESENT || Load::isLocal(loadEntry))
479 // Local load: check physical pin in local override bitmask
480 ? (Shared::overrideBitmask & (1U << Load::pinOf(loadEntry))) != 0
481 // Remote load: check its ordinal in the remote override bitmask
482 : (Shared::remoteOverrideBitmask & (1U << Load::remoteOrdinal(physicalLoadPin, iLoad))) != 0
483 };
484
485 const bool bOn{ bOverrideActive || (bDiversionEnabled && (loadPrioritiesAndState[idx] & loadStateOnBit)) };
486 physicalLoadState[iLoad] = (!CALIBRATION_MODE && bOn) ? LoadStates::LOAD_ON : LoadStates::LOAD_OFF;
487 } while (idx);
488}

507void processPolarity(const uint8_t phase, const uint16_t rawSample)
508{
509 // With left-aligned ADC and Q16.16 bias, simply subtract the integer part
510 // rawSample is already left-aligned (16-bit), extract bias from Q16.16 (upper 16 bits)
511 // Add 0.5 for the rounding, +32 when 10-bits left aligned `1<<(15-10)`
512 i_sampleVminusDC[phase] = (rawSample | 32U) - i_DCoffset_V[phase]; // +0.5 for the rounding
513
514 polarityOfMostRecentSampleV[phase] = (i_sampleVminusDC[phase] > 0) ? Polarities::POSITIVE : Polarities::NEGATIVE;
515}

534void processCurrentRawSample(const uint8_t phase, const uint16_t rawSample)
535{
536 // extra items for an LPF to improve the processing of data samples from CT1
537 static int32_t lpf_long[NO_OF_PHASES]{}; // new LPF, for offsetting the behaviour of CTx as a HPF
538
539 // remove most of the DC offset from the current sample.
540 // Use the voltage bias, it's most likely more accurate than the default mid-range.
541 // Add 0.5 for the rounding, +32 when 10-bits left aligned `1<<(15-10)`
542 int16_t sampleIminusDC = (rawSample | 32U) - i_DCoffset_V[phase];
543
544 // extra filtering to offset the HPF effect of CTx
545 // Using if constexpr ensures zero overhead when CT filtering is disabled
546 if constexpr (lpf_gain != 0.0F && alpha != 0.0F)
547 {
548 const auto last_lpf_long{ lpf_long[phase] };
549 lpf_long[phase] += alpha * (sampleIminusDC - last_lpf_long);
550 sampleIminusDC += (lpf_gain * lpf_long[phase]);
551 }
552
553 // calculate the "real power" in this sample pair and add to the accumulated sum
554 // Reduce to 14-bits (x16 scale) before multiplication for efficiency on 8-bit MCU
555 const int16_t filtV_div4{ i_sampleVminusDC[phase] >> 2 }; // reduce to 14-bits (now x16, or 2^4)
556 const int16_t filtI_div4{ sampleIminusDC >> 2 }; // reduce to 14-bits (now x16, or 2^4)
557
558 // Using optimized assembly multiplication for ~3x speedup in ISR
559 int32_t instP;
560 multS16x16_to32(instP, filtV_div4, filtI_div4); // 32-bits (now x256, or 2^8)
561 instP >>= 8; // scaling is now x1, as for Mk2 (V_ADC x I_ADC)
562
563 l_sumP[phase] += instP; // cumulative power, scaling as for Mk2 (V_ADC x I_ADC)
564 l_sumP_atSupplyPoint[phase] += instP; // cumulative power, scaling as for Mk2 (V_ADC x I_ADC)
565}

617void processVoltage(const uint8_t phase)
618{
619 // for the Vrms calculation (for datalogging only)
620 // Reduce to 14-bits (x16 scale) before multiplication for efficiency on 8-bit MCU
621 const int16_t filtV_div4{ i_sampleVminusDC[phase] >> 2 }; // reduce to 14-bits (now x16, or 2^4)
622
623 // Using optimized assembly multiplication for ~3x speedup in ISR
624 int32_t inst_Vsquared_signed;
625 multS16x16_to32(inst_Vsquared_signed, filtV_div4, filtV_div4);
626 uint32_t inst_Vsquared = static_cast< uint32_t >(inst_Vsquared_signed);
627
628 // Shift strategy verified safe by unit tests for all datalog periods and voltages.
629 // Datalog::vrmsFactor() compensates the x1/16 scaling: keep both conditions the same.
630 if constexpr (DATALOG_PERIOD_IN_SECONDS > 10)
631 {
632 inst_Vsquared >>= 12; // scaling is now x1/16 (V_ADC x V_ADC)
633 }
634 else
635 {
636 inst_Vsquared >>= 8; // scaling is now x1 (V_ADC x V_ADC)
637 }
638
639 l_sum_Vsquared[phase] += inst_Vsquared; // cumulative V^2 (V_ADC x V_ADC)
640 //
641 // store items for use during next loop
642 l_filterDC_V[phase] += i_sampleVminusDC[phase]; // for use with LP filter
643 polarityConfirmedOfLastSampleV[phase] = polarityConfirmed[phase]; // for identification of half cycle boundaries
644 ++n_samplesDuringThisMainsCycle[phase]; // for real power calculations
645}

695void proceedHighEnergyLevel()
696{
697 bool bOK_toAddLoad{ true };
698 const auto tempLoad{ nextLogicalLoadToBeAdded() };
699
700 if (tempLoad == NO_OF_DUMPLOADS)
701 {
702 return;
703 }
704
705 // a load which is now OFF has been identified for potentially being switched ON
706 if (b_recentTransition)
707 {
708 // During the post-transition period, any increase in the energy level is noted.
709 l_upperEnergyThreshold = l_energyInBucket_main;
710
711 // the energy thresholds must remain within range
712 if (l_upperEnergyThreshold > l_capacityOfEnergyBucket_main)
713 {
714 l_upperEnergyThreshold = l_capacityOfEnergyBucket_main;
715 }
716
717 // Only the active load may be switched during this period. All other loads must
718 // wait until the recent transition has had sufficient opportunity to take effect.
719 bOK_toAddLoad = (tempLoad == activeLoad);
720 }
721
722 if (bOK_toAddLoad)
723 {
724 loadPrioritiesAndState[tempLoad] |= loadStateOnBit;
725 activeLoad = tempLoad;
726 postTransitionCount = 0;
727 b_recentTransition = true;
728 }
729}

795void processStartNewCycle()
796{
797 // Restrictions apply for the period immediately after a load has been switched.
798 // Here the b_recentTransition flag is checked and updated as necessary.
799 // if (b_recentTransition)
800 // b_recentTransition = (++postTransitionCount < POST_TRANSITION_MAX_COUNT);
801 // for optimization, the next line is equivalent to the two lines above
802 b_recentTransition &= (++postTransitionCount < POST_TRANSITION_MAX_COUNT);
803
804 if (l_energyInBucket_main > l_midPointOfEnergyBucket_main)
805 {
806 // the energy state is in the upper half of the working range
807 l_lowerEnergyThreshold = l_lowerThreshold_default; // reset the "opposite" threshold
808 if (l_energyInBucket_main > l_upperEnergyThreshold)
809 {
810 // Because the energy level is high, some action may be required
811 proceedHighEnergyLevel();
812 }
813 }
814 else
815 {
816 // the energy state is in the lower half of the working range
817 l_upperEnergyThreshold = l_upperThreshold_default; // reset the "opposite" threshold
818 if (l_energyInBucket_main < l_lowerEnergyThreshold)
819 {
820 // Because the energy level is low, some action may be required
821 proceedLowEnergyLevel();
822 }
823 }
824
825 updatePhysicalLoadStates(); // allows the logical-to-physical mapping to be changed
826
827 updatePortsStates(); // update the control ports for each of the physical loads
828
829 if constexpr (REMOTE_LOADS_PRESENT)
830 {
831 // Rebuild the RF payloads AFTER local physical ports are updated.
832 // The actual transmission happens in loop(), never here.
833 remoteLoads.updateLoads< physicalLoadPin >(physicalLoadState);
834 }
835
836 if (loadPrioritiesAndState[0] & loadStateOnBit)
837 {
838 absenceOfDivertedEnergyCountInMC = 0;
839 }
840 else
841 {
842 ++absenceOfDivertedEnergyCountInMC;
843 }
844
845 // Now that the energy-related decisions have been taken, min and max limits can now
846 // be applied to the level of the energy bucket. This is to ensure correct operation
847 // when conditions change, i.e. when import changes to export, and vice versa.
848 //
849 if (l_energyInBucket_main > l_capacityOfEnergyBucket_main)
850 {
851 l_energyInBucket_main = l_capacityOfEnergyBucket_main;
852 }
853 else if (l_energyInBucket_main < 0)
854 {
855 l_energyInBucket_main = 0;
856 }
857}

952void processLatestContribution(const uint8_t phase)
953{
954 // for efficiency, the energy scale is Joules * SUPPLY_FREQUENCY, in integer fixed point
955 // add the latest energy contribution to the main energy accumulator
956 // (sumP / n) x cal == sumP x (cal / n): cal / n comes from the table, so there is no division
957 const uint8_t n{ n_samplesDuringThisMainsCycle[phase] };
958 int32_t sumP{ l_sumP[phase] };
959 uint8_t index{ static_cast< uint8_t >(n - N_MIN) };
960
961 if ((n < N_MIN) || (n > N_MAX))
962 {
963 // unexpected cycle length (start-up, missing phase): rescale the sum to N_MIN samples
964 sumP = (sumP / n) * N_MIN;
965 index = 0;
966 }
967
968 l_energyInBucket_main += Energy::contribution(sumP, pgm_read_word(&powerCalPerSample.value[phase][index]), powerCalPerSampleShift);
969
970 // apply any adjustment that is required.
971 if (0 == phase)
972 {
