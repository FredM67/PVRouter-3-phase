91// ANALOG pins:

94// DIGITAL pins:
95// - D0 & D1: Reserved for the Serial interface.

122// EWMA Filter Tuning for Cloud Immunity
123//
124// The RELAY_FILTER_DELAY parameter controls how aggressively the EWMA filter smooths
125// power measurements before making relay decisions. This directly affects cloud immunity vs responsiveness.
126//
127// 🌤️ Quick Reference by Climate:
128// - Clear sky regions (desert, dry): 1 minute (fast response, minimal clouds)
129// - Mixed conditions (most locations): 2 minutes (recommended default)
130// - Frequently cloudy (coastal): 3 minutes (enhanced stability)
131// - Very cloudy (mountain, tropical): 4 minutes (maximum stability)
132//
133// 🧪 Scientific Tuning:
134// Run cloud pattern analysis tests to determine optimal setting for your climate:
135// cd Mk2_3phase_RFdatalog_temp && pio test -e native --filter="test_cloud_patterns" -v
136//
137// 📖 Full guide: docs/Cloud_Pattern_Tuning_Guide.md

153// This is an example of override pin configuration.
154// You can modify the pin numbers and associated loads/relays as needed.
155// Ensure that the pins used do not conflict with other functionalities in your setup.
156//
157// Helper functions available:
158// LOAD(n) - Returns the pin for load n of physicalLoadPin[] (physical if the
159// load is local, virtual >= REMOTE_PIN_BASE if it is remote)
160// LOCAL_LOAD(n) - Same as LOAD(n), for a load known to be local
161// REMOTE_LOAD(n) - Returns the virtual pin for the n-th REMOTE load (>= 128).
162// n counts remote loads only, in physicalLoadPin[] order.
163// ALL_LOCAL_LOADS() - uint32_t bitmask, lower 16 bits for local load pins
164// ALL_REMOTE_LOADS()- uint32_t bitmask, upper 16 bits for remote loads (bit 16 = remote 0)
165// ALL_LOADS() - uint32_t combining local (lower 16 bits) and remote (upper 16 bits)
166// RELAY(n) - Returns pin for relay n
167// ALL_RELAYS() - uint32_t bitmask, lower 16 bits for relay pins
168// ALL_LOADS_AND_RELAYS() - uint32_t combining all loads and relays
169//
170// Example configurations:
171// inline constexpr OverridePins overridePins{
172// { { 4, ALL_LOADS() }, // Control all loads (local + remote)
173// { 5, ALL_LOCAL_LOADS() }, // Control only local loads
174// { 6, ALL_REMOTE_LOADS() }, // Control only remote loads
175// { 7, { LOAD(0), REMOTE_LOAD(1) } }, // Mixed: load 0 + the 2nd remote load
176// { 8, { LOAD(0), LOAD(1), LOAD(2), LOAD(3) } }, // Using LOAD() for any load index
177// { 9, ALL_LOADS_AND_RELAYS() } } }; // All loads and relays
178
